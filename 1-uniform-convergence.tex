\section{Uniform Convergence}

This section concerns a sequence of \emph{functions}
\[
    \functiondomain{f_n}{E}{\RR}, \quad n = 1, 2, 3, \ldots,
\]
where \(E \subseteq \RR\), and the question of the \emph{convergence} of the sequence \(\pqty{f_n}\).

\subsection{Pointwise Convergence}

\begin{definition}[Pointwise Convergence]
    Let \(\functiondomain{f_n, f}{E}{\RR}\) for \(n \in \NN\). We say that the sequence \(\pqty{f_n}\) \term{converges pointwise} to \(f\) on \(E\), if for every \(x \in E\), the sequence \(\pqty{f_n(x)}\) converges to \(f(x)\), i.e.,
    \[
        f\pqty{x} = \lim_{n \to \infty} f_n\pqty{x}, \quad \forall x \in E.
    \]
    
    We say \(f\) is the \term{pointwise limit} of \(\pqty{f_n}\), and denote \(f_n \to f\) pointwise in \(E\).
\end{definition}

We would like to deduce properties such as continuity, differentiability, and integrability for the limit \(f\), from those properties for \(f_n\)s. However, pointwise convergence is \emph{not} strong enough to guarantee such property.

\begin{example}[Pointwise Convergence does not Preserve Continuity]
    Suppose \(\functiondomain{f_n}{\ccintv{-1}{1}}{\RR}\), defined by
    \[
        f_n\pqty{x} = x^{\flatfrac{1}{\pqty{2n + 1}}}
    \]
    for all \(n \in \NN\). Then \(f_n\) is continuous for all \(n \in \NN\). However, note that
    \[
        f_n\pqty{x} \to f\pqty{x} = \begin{cases}
            1 & \qif 0 < x < 1, \\
            0 & \qif x = 0, \\
            -1 & \qif -1 < x < 0, \\
        \end{cases}
    \]
    pointwise on \(\ccintv{-1}{1}\), so the pointwise limit is not continuous. This is illustrated in \autoref{fig:pointwise-limit}.
\end{example}

\begin{example}[Pointwise Convergence does not Preserve Integrability]
    Let \(q_1, q_2, q_3, \ldots\) be an enumeration of \(\QQ \cap \ccintv{0}{1}\). Define \(\functiondomain{f_n}{\ccintv{0}{1}}{\RR}\) by
    \[
        f_n\pqty{x} = \begin{cases}
            1 & \qif x \in \Bqty{q_1, q_2, \ldots, q_n}, \\
            0 & \qif x \in \ccintv{0}{1} \setminus \Bqty{q_1, q_2, \ldots, q_n}.
        \end{cases}
    \]

    Note that \(f_n\) is integrable for all \(n \in \NN\), since it is continuous apart from points in \(\Bqty{q_1, q_2, \ldots. q_n}\).
    
    However, the pointwise limit of \(\pqty{f_n}\) is
    \[
        f\pqty{x} = \lim_{n \to \infty} f_n\pqty{x} = \indicator_{\QQ} \pqty{x}.
    \]
    which is not integrable.
\end{example}

A natural question that continues from integrability, is that if \(f_n\) is integrable for all \(n \in \NN\), does it follow that their pointwise limit \(f\), given integrable, integrates to the limit of the integrals?

    \centering
    \includegraphics[width=0.7\linewidth]{fig/pointwise-limit-integral.png}
    \caption{Function in \autoref{ex:pointwise-limit-integral}}
    \label{fig:pointwise-limit-integral}%
\end{figure}

\begin{example}[Pointwise Limit does not Preserve Integral]
    \label{ex:pointwise-limit-integral}
    Consider the functions defined on \(\ccintv{0}{1}\) where
    \[
        f_n\pqty{x} = \begin{cases}
            n^2 x &\qif n \in \ccintv{0}{\flatfrac{1}{n}}, \\
            n^2 \pqty{\flatfrac{2}{n} - x} &\qif n \in \ccintv{\flatfrac{1}{n}}{\flatfrac{2}{n}}, \\
            0 &\qif x \in \ccintv{\flatfrac{2}{n}}{1}.
        \end{cases}
    \]

    Then clearly its pointwise limit is identically zero, since the triangle eventually moves to the left of any \(x\), however small (apart from \(x = 0\) which is identically zero), and hence
    \[
        \int_{0}^{1} f\pqty{x} \dd{x} = 0.
    \]

    However, notice that
    \[
        \int_{0}^{1} f_n\pqty{x} \dd{x} = 1, \quad \forall n \in \NN
    \]
    and hence
    \[
        \lim_{n \to \infty} \int_{0}^{1} f_n\pqty{x} \dd{x} = 1 \neq 0 = \int_{0}^{1} f\pqty{x} \dd{x} = \int_{0}^{1} \lim_{n \to \infty} f_n \pqty{x} \dd{x}
    \]
\end{example}

\subsection{Uniform Convergence}

We now give a stronger notion of convergence which preserves continuity and integrability. But clearly, we do not want too strong a notion of convergence, for example, the trivial one that \(f_n \to f\) means \(f = f_n\) for sufficiently large \(n\).

\begin{definition}[Uniform Convergence]
    Let \(\functiondomain{f_n, f}{E}{\RR}\) for \(n \in \NN\). We say that the sequence \(\pqty{f_n}\) \term{converges uniformly} to \(f\) on \(E\), if
    \[
        \forall \epsilon > 0, \exists N = N\pqty{\epsilon} \in \NN, \forall n \geq N, \forall x \in E: \abs{f_n\pqty{x} - f\pqty{x}} < \epsilon.
    \]

    This is equivalent to saying
    \[
        \forall \epsilon > 0, \exists N = N\pqty{\epsilon} \in \NN: n \geq N \implies \sup_{x \in E} \abs{f_n\pqty{x} - f\pqty{x}} < \epsilon,
    \]
    which in limit notation, reads
    \[
        \lim_{n \to \infty} \sup_{x \in E} \abs{f_n\pqty{x} - f\pqty{x}} = 0.
    \]

    We say \(f\) is the \term{uniform limit} of \(\pqty{f_n}\), and denote \(f_n \to f\) uniformly in \(E\).
\end{definition}

\begin{remark}
    Compare this with pointwise convergence, which states
    \[
        \forall x \in E, \forall \epsilon > 0, \exists N = N\pqty{\epsilon, x} \in \NN, \forall n \geq N: \abs{f_n\pqty{x} - f\pqty{x}} < \epsilon.
    \]

    Note that here, \(N\) may depend on \(x\). It is clear that uniform convergence implies pointwise convergence, while the converse is not true.
\end{remark}

\begin{example}[Pointwise Convergence does not imply Uniform Convergence]
    We look at the previous example, illustrated in \autoref{fig:pointwise-limit-integral}, where
        \[
            f_n\pqty{x} = \begin{cases}
                n^2 x &\qif n \in \ccintv{0}{\flatfrac{1}{n}}, \\
                n^2 \pqty{\flatfrac{2}{n} - x} &\qif n \in \ccintv{\flatfrac{1}{n}}{\flatfrac{2}{n}}, \\
                0 &\qif x \in \ccintv{\flatfrac{2}{n}}{1}.
            \end{cases}
        \]
        and \(f_n \to f\) where \(f\pqty{x} = 0\) \emph{pointwise} on \(\ccintv{0}{1}\). However, notice that
        \[
            \sup_{x \in \ccintv{0}{1}} \abs{f_n\pqty{x} - f\pqty{x}} = n \to \infty \neq 0,
        \]
        so \(f_n \notto f\) \emph{uniformly} on \(\ccintv{0}{1}\).
\end{example}

\begin{example}[Uniform Convergence depends on Domain]
    We look at the functions
    \[
        \function{f_n}{\RR}{\RR}{x}{\frac{x}{n}.}
    \]

    Then we have \(f_n \to 0\) \emph{pointwise} on \(\RR\), while
    \[
        \sup_{x \in \RR} \abs{f_n\pqty{x} - 0} = \infty,
    \]
    so \(f_n \notto 0\) \emph{uniformly} on \(\RR\).

    If we restrict the domain, we can achieve uniform convergence. Let \(a \in \oointv{0}{\plusinfty}\), \(\functiondomain{f_n}{\ccintv{-a}{a}}{\RR}\).

    Then
    \[
        \sup_{x \in \ccintv{-a}{a}} \abs{f_n\pqty{x}} = \sup_{x \in \ccintv{-a}{a}} \frac{\abs{x}}{n} = \frac{a}{n} \to 0,
    \]
    so \(f_n \to 0\) \emph{uniformly} on \(\ccintv{-a}{a}\).
\end{example}

Similarly to Cauchy sequences of real numbers, we can define a notion of Cauchy sequences of functions.

\subsection{Cauchy Criterion for Uniform Convergence}

\begin{definition}[Uniform Cauchy]
    Let \(\functiondomain{f_n}{E}{\RR}\) be a sequence of functions. We say \(\pqty{f_n}\) is \term{uniformly Cauchy} on \(E\), if
    \[
        \forall \epsilon > 0, \exists N = N\pqty{\epsilon} \in \NN: m, n \geq N \implies \sup_{x \in E} \abs{f_n\pqty{x} - f_m\pqty{x}} < \epsilon.
    \]
\end{definition}

\begin{theorem}[Cauchy Criterion for Uniform Convergence]
    \label{thm:cauchy-criterion-uniform-convergence}%
    Let \(\functiondomain{f_n}{E}{\RR}\) be a sequence of functions. Then \(\pqty{f_n}\) converges uniformly on \(E\) if and only if it is uniformly Cauchy on \(E\).
\end{theorem}

\begin{proof}
    Suppose \(\pqty{f_n}\) converges uniformly on \(E\) to some function \(f\). Let \(\epsilon > 0\), then there is some \(N = N\pqty{\epsilon} \in \NN\), such that for all \(n \geq N\)
    \[
        \sup_{x \in E} \abs{f_n\pqty{x} - f\pqty{x}} < \epsilon.
    \]

    Then for \(m, n \geq N\), we have
    \begin{align*}
        \abs{f_n\pqty{x} - f_m\pqty{x}} &\leq \abs{f_n\pqty{x} - f\pqty{x}} + \abs{f\pqty{x} - f_m\pqty{x}} \\
        &\leq \sup_{x \in E} \abs{f_n\pqty{x} - f\pqty{x}} + \sup_{x \in E} \abs{f\pqty{x} - f_m\pqty{x}},
    \end{align*}
    and hence
    \[
        \sup_{x \in E} \abs{f_n\pqty{x} - f_m\pqty{x}} \leq \sup_{x \in E} \abs{f_n\pqty{x} - f\pqty{x}} + \sup_{x \in E} \abs{f\pqty{x} - f_m\pqty{x}} < 2\epsilon.
    \]
    
    Now, suppose instead that \(\pqty{f_n}\) is uniformly Cauchy on \(E\). Then for all \(x \in E\), the sequence of real numbers \(\pqty{f_n \pqty{x}}\) is Cauchy. By Completeness of \(\RR\) (see \textit{IA Analysis I}), there exists a \emph{pointwise} limit \(f\pqty{x} = \lim_{n \to \infty} f_n\pqty{x}\) for all \(x \in E\). We now show that \(f_n \to f\) \emph{uniformly} on \(E\).

    To check that \(f_n \to f\) is uniform, let \(\epsilon > 0\). By hypothesis, there exists \(N\), such that for all \(m, n \geq N\), that
    \[
        \sup_{x \in E} \abs{f_n\pqty{x} - f_m\pqty{x}} < \epsilon.
    \]

    For a fixed \(x\), we have
    \begin{align*}
        \abs{f_n\pqty{x} - f\pqty{x}} &\leq \abs{f_n\pqty{x} - f_m\pqty{x}} + \abs{f_m\pqty{x} - f\pqty{x}} \\
        &< \epsilon + \abs{f_m\pqty{x} - f\pqty{x}} \\
        &< \epsilon + \epsilon = 2 \epsilon,
    \end{align*}
    if \(m\) is sufficiently large (since \(f_n \to f\) pointwise). Taking supremum over \(x \in E\), we have
    \[
        \sup_{x \in E} \abs{f_n\pqty{x} - f\pqty{x}} < 2 \epsilon,
    \]
    which shows that \(f_n \to f\) uniformly on \(E\).
\end{proof}

\subsection{Properties Preserved under Uniform Convergence}

\begin{theorem}[Uniform Convergence preserves Continuity at a Point]
    Let \(\functiondomain{f_n, f}{\ccintv{a}{b}}{\RR}\) and \(f_n \to f\) uniformly. Suppose that, for some \(x_0 \in \ccintv{a}{b}\), \(f_n\) is continuous at \(x_0\) for all \(n\). Then \(f\) is continuous at \(x_0\).
\end{theorem}

\begin{proof}
    Let \(\epsilon > 0\). By uniform convergence, there is some \(N\), such that for all \(n \geq N\), we have
    \[
        \sup_{x \in \ccintv{a}{b}} \abs{f_n\pqty{x} - f\pqty{x}} < \epsilon.
    \]

    Then we have
    \begin{align*}
        \abs{f\pqty{y} - f\pqty{x}} &\leq \abs{f\pqty{y} - f_N\pqty{y}} + \abs{f_N\pqty{y} - f_N\pqty{x}} + \abs{f_N\pqty{x} - f\pqty{x}} \\
        &< 2\epsilon + \abs{f_N\pqty{y} - f_N\pqty{x}}.
    \end{align*}

    Take \(x = x_0\). If \(f_N\pqty{x}\) is continuous at \(x_0\), then for sufficiently small \(\delta = \delta\pqty{x_0, epsilon} > 0\), such that for all \(\abs{y - x_0 < \delta}\), we have
    \[
        \abs{f_N\pqty{y} - f_N\pqty{x_0}} < \epsilon,
    \]
    by continuity of \(f_N\) at \(x_0\). Hence, for all \(\abs{y - x_0} < \delta\), we have
    \[
        \abs{f\pqty{y} - f\pqty{x_0}} < 3 \epsilon,
    \]
    which shows that \(f\) is continuous at \(x_0\).
\end{proof}

Applying this to all \(x_0 \in \ccintv{a}{b}\) gives the following result.

\begin{corollary}[Uniform Convergence preserves Continuity]
    \label{cor:uniform-convergence-preserves-continuity}%
    Let \(\functiondomain{f_n, f}{\ccintv{a}{b}}{\RR}\) and \(f_n \to f\) uniformly. Suppose that, \(f_n\) is continuous on \(\ccintv{a}{b}\) for all \(n\). Then \(f\) is continuous on \(\ccintv{a}{b}\).
\end{corollary}

Recall from \textit{IA Analysis I} that \(\contClass{}\pqty{\ccintv{a}{b}}\) denotes the space of functions continuous on \(\ccintv{a}{b}\). Back in \textit{IA Analysis I}, recall that restricting to \(\QQ\) does \emph{not} preserve completeness of \(\RR\), since there are rational Cauchy sequences that do not converge in \(\QQ\). However, this space above is complete when we talk about uniform convergence (i.e. measuring `distance' using the supremum).

\begin{theorem}[Completeness of \(\contClass{}\pqty{\ccintv{a}{b}}\)]
    Let \(\pqty{f_n}\) be a uniformly Cauchy sequence in \(\contClass{}\pqty{\ccintv{a}{b}}\). Then there exists a continuous function \(f \in \contClass{}\pqty{\ccintv{a}{b}}\), such that \(f_n \to f\) uniformly on \(\ccintv{a}{b}\).
\end{theorem}

\begin{proof}
    By \autoref{thm:cauchy-criterion-uniform-convergence}, there exists a function \(f\) such that \(f_n \to f\) uniformly on \(\ccintv{a}{b}\). By \autoref{cor:uniform-convergence-preserves-continuity}, \(f\) is continuous on \(\ccintv{a}{b}\), so \(f \in \contClass{}\pqty{\ccintv{a}{b}}\).
\end{proof}

Integrals also behave nicely under uniform convergence.

\begin{theorem}[Convergence of Integrals]
    \label{thm:uniform-convergence-integrals}%
    Let \(\functiondomain{f_n, f}{\ccintv{a}{b}}{\RR}\) be integrable and \(f_n \to f\) uniformly on \(\ccintv{a}{b}\). Then
    \[
        \lim_{n \to \infty} \int_a^b f_n(x) \dd{x} = \int_a^b f(x) \dd{x}.
    \]
\end{theorem}

\begin{remark}
    At the current state, we assumed that \(f\) is integrable. This is, however, redundant, as we will see in the future.
\end{remark}

\begin{proof}
    We note
    \begin{align*}
        \abs{\int_a^b f_n(x) \dd{x} - \int_a^b f(x) \dd{x}} &= \abs{\int_a^b \pqty{f_n(x) - f(x)} \dd{x}} \\
        &\leq \int_a^b \abs{f_n(x) - f(x)} \dd{x} \\
        &\leq \int_a^b \sup_{x \in \ccintv{a}{b}} \abs{f_n(x) - f(x)} \dd{x} \\
        &= \pqty{b - a} \sup_{x \in \ccintv{a}{b}} \abs{f_n(x) - f(x)},
    \end{align*}
    and this tends to zero as \(n \to \infty\) by uniform convergence.
\end{proof}

Differentiation, however, is a subtle matter, and uniform convergence does not necessarily preserve differentiability. Let \(\functiondomain{f_n, f}{\ccintv{a}{b}}{\RR}\), with \(f_n \to f\) uniformly on \(\ccintv{a}{b}\). Even if \(f_n\) is differentiable for all \(n\), neither of the following need to be true:
\begin{enumerate}[label=(\roman*)]
    \item that \(f\) is differentiable, or
    \item given \(f\) differentiable, that \(f_n' \to f'\), even pointwise.
\end{enumerate}

\begin{example}[Differentiable Functions with Non-Differentiable Uniform Limit]
    Consider the functions defined on \(\ccintv{-1}{1}\) where
    \[
        \function{f_n}{\ccintv{-1}{1}}{\RR}{x}{\abs{x}^{1 + \flatfrac{1}{n}}.}
    \]

    Then \(f_n\) is differentiable on \(\ccintv{-1}{1}\). In particular, at \(x = 0\),
    \[
        \lim_{h \to 0} \frac{f_n\pqty{h} - f_n\pqty{0}}{h} = \lim_{h \to 0} \sign\pqty{h} \abs{h}^{\frac{1}{n}} \to 0,
    \]
    showing \(f'\pqty{0} = 0\).

    Note \(f_n \to f\) uniformly on \(\ccintv{-1}{1}\), where \(f\pqty{x} = \abs{x}\), which is not differentiable at \(x = 0\).
\end{example}

\begin{example}[Differentiable Functions with Non-Converging Derivative]
    Consider the functions
    \[
        \function{f_n}{\RR}{\RR}{x}{\frac{\sin\pqty{nx}}{\sqrt{n}}.}
    \]

    Then clearly
    \[
        \sup_{x \in \RR} \abs{f_n\pqty{x}} = \frac{1}{\sqrt{n}} \to 0,
    \]
    so \(f_n \to f\) uniformly on \(\RR\), where \(f\pqty{x} = 0\). However, note that
    \[
        f_n'\pqty{x} = \sqrt{n} \cos\pqty{nx},
    \]
    and clearly this does not converge to \(f'\pqty{x} = 0\) at \(x = 0\).
\end{example}

\begin{theorem}[Uniform Convergence and Differentiability]
    \label{thm:uniform-convergence-differentiability}%
    Let \(\functiondomain{f_n}{\ccintv{a}{b}}{\RR}\) be differentiable on \(\ccintv{a}{b}\), such that:
    \begin{enumerate}[label=(\roman*)]
        \item \(\pqty{f_n'}\) converges uniformly to some function \(\functiondomain{g}{\ccintv{a}{b}}{\RR}\), and
        \item there exists \(c \in \ccintv{a}{b}\) such that \(\pqty{f_n\pqty{c}}\) converges.
    \end{enumerate}

    Then \(\pqty{f_n}\) converges uniformly to some function \(\functiondomain{f}{\ccintv{a}{b}}{\RR}\), and \(f\) is differentiable on \(\ccintv{a}{b}\) with \(f' = g\), i.e.,
    \[
        \pqty{\lim_{n \to \infty} f_n}' = \lim_{n \to \infty} f_n'
    \]
    with uniform convergence on \(\ccintv{a}{b}\).
\end{theorem}

\begin{remarks}
    \begin{itemize}
        \item We did not assume \(f_n'\) to be continuous, or even integrable. For example, consider the function
        \[
            f\pqty{x} = \begin{cases}
                x^2 \sin \pqty{\dfrac{1}{x}}, & x \neq 0, \\
                0, & x = 0.
            \end{cases}
        \]
        with derivative
        \[
            f'\pqty{x} = \begin{cases}
                2x \sin \pqty{\dfrac{1}{x}} - \cos\pqty{\dfrac{1}{x}}, & x \neq 0, \\
                0, & x = 0.
            \end{cases}
        \]

        Note \(f'\) is not continuous at \(x = 0\).

        Aside, there are functions which are differentiable with bounded derivatives everywhere, but not \emph{Riemann} integrable, for example, the Volterra function.
        \item If we assume further that \(f_n'\) are continuous, then the proof is easier utilising the Fundamental Theorem of Calculus (recall from \textit{IA Analysis I}).
    \end{itemize}
\end{remarks}

\begin{proof}
    To show that \(\pqty{f_n}\) converges uniformly, we first show that \(\pqty{f_n}\) is uniformly Cauchy. For any \(x \in \ccintv{a}{b}\), by the mean value theorem,
    \[
        f_n\pqty{x} - f_m\pqty{x} = f_n\pqty{c} - f_m\pqty{c} + \pqty{x - c} \pqty{f_n'\pqty{x_{m, n}} - f_m'\pqty{x_{m, n}}}
    \]
    for some \(x_{m, n}\) between \(c\) and \(x\). Then
    \[
        \sup_{\ccintv{a}{b}} \abs{f_n - f_m} \leq \abs{f_n\pqty{c} - f_m\pqty{c}} + \pqty{b - a} \sup_{\ccintv{a}{b}} \abs{f_n' - f_m'} \to 0
    \]
    so \(f_n\) is uniformly Cauchy, and hence converges uniformly, to some function \(f\), by \autoref{thm:cauchy-criterion-uniform-convergence}.

    To check that \(f\) is differentiable with \(f' = g\) in \(\ccintv{a}{b}\), fix some \(y \in \ccintv{a}{b}\), and define
    \[
        h\pqty{x} = \begin{cases}
            \frac{f\pqty{x} - f\pqty{y}}{x - y}, & x \neq y, \\
            g\pqty{y}, & x = y.
        \end{cases}
    \]

    So \(f'\pqty{y} = g\pqty{y}\), if and only if \(h\) is continuous at \(y\).

    Consider the sequence of functions
    \[
        h_n\pqty{x} = \begin{cases}
            \frac{f_n\pqty{x} - f_n\pqty{y}}{x - y}, & x \neq y, \\
            f_n'\pqty{y}, & x = y.
        \end{cases}
    \]

    \(h_n\) is continuous for all \(n\), since \(f_n\) is differentiable. Furthermore, \(h_n \to h\) pointwise. If we can show \(\pqty{h_n}\) is Cauchy, then this will imply the convergence is uniform, hence finishing the proof.

    We have
    \begin{align*}
        h_n\pqty{x} - h_m\pqty{x} &= \begin{cases}
            \frac{f_n\pqty{x} - f_m\pqty{x}}{x - y} - \frac{f_n\pqty{y} - f_m\pqty{y}}{x - y}, & x \neq y, \\
            f_n'\pqty{y} - f_m'\pqty{y}, & x = y,
        \end{cases} \\
        &= \begin{cases}
            \frac{f_n\pqty{x} - f_m\pqty{x}}{x - y} - \frac{f_n\pqty{y} - f_m\pqty{y}}{x - y}, & x \neq y, \\
            f_n'\pqty{y} - f_m'\pqty{y}, & x = y,
        \end{cases} \\
        &= \begin{cases}
            \pqty{f_n - f_m}' \pqty{y_{m, n}}, & x \neq y, \\
            f_n'\pqty{y} - f_m'\pqty{y}, & x = y,
        \end{cases}
    \end{align*}
    for some \(y_{m, n}\) between \(x\) and \(y\), by the Mean Value Theorem. Hence, we have
    \[
        \sup_{\ccintv{a}{b}} \abs{h_n - h_m} \leq \sup_{\ccintv{a}{b}} \abs{f_n' - f_m'} \to 0,
    \]
    so \(h_n\) is uniformly Cauchy, and hence converges uniformly to \(h\). By \autoref{cor:uniform-convergence-preserves-continuity} (Uniform Convergence preserves Continuity), \(h\) is continuous at \(y\), and hence \(f'\pqty{y} = g\pqty{y}\), finishing our proof.
\end{proof}

\begin{proof}[\proofname\ if \(f_n'\) are continuous]
    Under this assumption, we have
    \[
        f_n\pqty{x} = f_n\pqty{c} + \int_{c}^{x} f_n'\pqty{t} \dd{t}
    \]
    for any \(x \in \ccintv{a}{b}\).

    If we look at the difference between two functions, we have
    \begin{align*}
        \abs{f_n\pqty{x} - f_m\pqty{x}} &\leq \abs{f_n\pqty{c} - f_m\pqty{c}} + \abs{\int_{c}^{x} \pqty{f_n' \pqty{t} - f_m'\pqty{t}} \dd{t}} \\
        &\leq \abs{f_n\pqty{c} - f_m\pqty{c}} + \pqty{b - a} \sup_{t \in \ccintv{a}{b}} \abs{f_n'\pqty{t} - f_m'\pqty{t}}
    \end{align*}
    and so the uniform Cauchy property of \(f_n\) follows from the Cauchy property of the real sequence \(\Bqty{f_n\pqty{c}}\), and the uniform Cauchy property of \(\pqty{f_n'}\).

    Let \(f\) be the uniform limit of \(\pqty{f_n}\), and \(g\) be the uniform limit of \(\pqty{f_n'}\), continuous.

    So \(g\) is integrable, and we have
    \[
        \int_{c}^{x} f_n'\pqty{t} \dd{t} \to \int_{c}^{x} g\pqty{t} \dd{t}.
    \]

    Therefore,
    \[
        f\pqty{x} = f\pqty{c} + \int_{c}^{x} g\pqty{t} \dd{t},
    \]
    and by continuity of \(g\), \(f\) is differentiable with \(f' = g\).
\end{proof}

The following proposition is very direct consequences from the definition of uniform convergence, and is left as an exercise to the reader.

\begin{proposition}[Algebra of Uniform Convergence]
    Let \(\functiondomain{f_n, g_n}{E}{\RR}\) for \(n \in \NN\).
    \begin{enumerate}[label=(\roman*)]
        \item If \(f_n \to f\), \(g_n \to g\) uniformly on \(E\), then \(f_n + g_n \to f + g\) uniformly on \(E\).
        \item If \(f_n \to f\) uniformly on \(E\), and \(\functiondomain{h}{E}{\RR}\) is bounded, then \(h f_n \to h f\) uniformly on \(E\).
    \end{enumerate}
\end{proposition}

\begin{examples}[Non-Bounded Function Multiplied by Uniformly Convergent Sequence]
    \begin{itemize}
        \item Let \(\functiondomain{f_n, h}{\RR}{\RR}\) with \(f_n \pqty{x} = \frac{1}{n}\), so \(f_n \to 0\)uniformly on \(\RR\).

        But if \(h\pqty{x} = x\), then \(\pqty{h f_n} \pqty{x} = \frac{x}{n}\) only converges pointwise to \(0\) on \(\RR\), but not uniformly.

        \item Let \(\functiondomain{f_n}{\oointv{0}{1}}{\RR}\) with \(f_n = \flatfrac{1}{n} \to 0\) uniformly.
        
        Take \(h\pqty{x} = \frac{1}{x}\), then \(\pqty{h f_n} \pqty{x} = \frac{1}{nx}\), and \[
            \sup_{x \in \oointv{0}{1}} \abs{h f_n\pqty{x} - 0} = \sup_{x \in \oointv{0}{1}} \frac{1}{nx} = \infty,
        \]
        so \(\pqty{h f_n}\) does not converge uniformly to \(0\) on \(\oointv{0}{1}\).
    \end{itemize}
\end{examples}

\subsection{Series of Functions}

\begin{definition}[Partial Sum of a Series of Functions]
    Let \(\functiondomain{g_n}{E}{\RR}\). Define \(\functiondomain{f_n}{E}{\RR}\) as
    \[
        f_n \pqty{x} = \sum_{j = 1}^{n} g_j \pqty{x}
    \]
    as the \term{\(n\)th partial sum}.
\end{definition}

\begin{definition}[Convergence of Series of Functions]
    We define the following.

    \begin{enumerate}[label=(\alph*)]
        \item We say that the series of functions \(\sum_{j = 1}^{\infty} g_j\) \term{converges at \(x \in E\)} if the sequence \(\pqty{f_n \pqty{x}}\) converges.
        \item The series \(\sum_{j = 1}^{\infty} g_j\) \term{converges uniformly} in \(E\) if the partial sum \(\pqty{f_n}\) converges uniformly in \(E\).
        \item The series \(\sum_{j = 1}^{\infty} g_j\) \term{converges absolutely at \(x \in E\)} if the sequence \(\sum_{j = 1}^{\infty} \abs{g_j \pqty{x}}\) converges.
        \item The series \(\sum_{j = 1}^{\infty} g_j\) \term{converges absolutely uniformly} in \(E\) if the partial sum \(\sum_{j = 1}^{n} \abs{g_j \pqty{x}}\) converges uniformly in \(E\).
    \end{enumerate}
\end{definition}

Recall for real-number series, absolute convergence implies convergence. This extends to series of functions as follows.

\begin{proposition}[Absolute Uniform Convergence implies Uniform Convergence]
    Let \(\functiondomain{g_n}{E}{\RR}\). If \(\sum_{j = 1}^{\infty} g_j\) converges absolutely uniformly in \(E\), then it converges uniformly in \(E\).
\end{proposition}

\begin{proof}
    Let \(f_N \pqty{x} = \sum_{j = 1}^{n} g_i \pqty{x}\). For \(n > m\), we have
    \[
        f_n \pqty{x} - f_m \pqty{x} = \sum_{j = m + 1}^{n} g_j \pqty{x},
    \]
    and hence
    \[
        \abs{f_n \pqty{x} - f_m \pqty{x}} \leq \sum_{j = m + 1}^{n} \abs{g_j \pqty{x}} = \sum_{j = 1}^{n} \abs{g_j \pqty{x}} - \sum_{j = 1}^{m} \abs{g_j \pqty{x}} = h_n \pqty{x} - h_m\pqty{x}
    \]
    which is the difference of the corresponding absolute value series
    \[
        h_n \pqty{x} = \sum_{j = 1}^{n} \abs{g_j \pqty{x}}.
    \]

    Therefore,
    \[
        \abs{f_n\pqty{x} - f_m\pqty{x}} \leq \abs{h_n\pqty{x} - h_m\pqty{x}}
    \]
    and uniform convergence is immediate by \autoref{thm:cauchy-criterion-uniform-convergence} (Cauchy criterion).
\end{proof}

\begin{remark}
    It is important to note that \textbf{uniform convergence + absolute convergence at every point does not imply absolute uniform convergence}.
\end{remark}

\begin{example}[Uniform Convergence and Absolute Convergence Pointwise]
    The series
    \[
        \sum_{j = 1}^{\infty} \frac{\pqty{-1}^j}{j} x^j
    \]
    for \(x \in \oointv{0}{1}\) converges absolutely at every \(x \in \oointv{0}{1}\), and converges uniformly on \(\oointv{0}{1}\). However, it does not converge absolutely uniformly on \(\oointv{0}{1}\), as we notice
    \[
        \sum_{j = 1}^{n} \abs{\frac{\pqty{-1}^j}{j} x^j} - \sum_{j = 1}^{m} \abs{\frac{\pqty{-1}^j}{j} x^j} = \sum_{j = m + 1}^{n} \frac{1}{j} x^j \geq \sum_{j = m + 1}^{n} \frac{1}{j} x^n.
    \]

    Taking the supremum over \(\oointv{0}{1}\), this is at least
    \[
        \sup_{x \in \oointv{0}{1}} \pqty{\sum_{j = 1}^{n} \abs{\frac{\pqty{-1}^j}{j} x^j} - \sum_{j = 1}^{m} \abs{\frac{\pqty{-1}^j}{j} x^j}} \geq \sum_{j = n + 1}^{m} \frac{1}{j}
    \]
    but this goes to infinity as \(n \to \infty\), for any fixed \(m\).
\end{example}

We have the following simple but very useful criterion for absolute uniform convergence.

\begin{theorem}[Weierstrass M-Test]
    \label{thm:weierstrass-m-test}%
    Let \(\functiondomain{g_j}{E}{\RR}\) for \(j \in \NN\). Suppose there exists numbers \(M_j \geq 0\) such that
    \[
        \sup_{x \in E} \abs{g_j\pqty{x}} \leq M_j
    \]
    and that the series \(\sum_{j = 1}^{\infty} M_j\) converges. Then the series \(\sum_{j = 1}^{\infty} g_j\) converges absolutely uniformly on \(E\).
\end{theorem}

\begin{proof}
    Taking the differences of the partial absolute series, we have
    \[
        \sum_{j = 1}^{n} \abs{g_j \pqty{x}} - \sum_{j = 1}^{m} \abs{h_j \pqty{x}} = \sum_{j = m + 1}^{n} \abs{g_j \pqty{x}} \leq \sum_{j = m + 1}^{n} M_j = \sum_{j = 1}^{n} M_j - \sum_{j = 1}^{m} M_j,
    \]
    and hence the supremum
    \[
        \sup_{x \in E} \abs{\sum_{j = 1}^{n} \abs{g_j \pqty{x}} - \sum_{j = 1}^{m} \abs{g_j \pqty{x}}} \leq \sum_{j = 1}^{n} M_j - \sum_{j = 1}^{m} M_j
    \]
    for all \(n, m\). Since the \(M_j\) series converges, the partial sums must be Cauchy, and hence the absolute partial sum series are uniformly Cauchy, and hence converge uniformly by \autoref{thm:cauchy-criterion-uniform-convergence} (Cauchy criterion).
\end{proof}

\subsection{Power Series}

\begin{theorem}[Radius of Convergence]
    \label{thm:radius-of-convergence}%
    Let
    \[
        \sum_{n = 0}^{\infty} c_n \pqty{x - a} ^n
    \]
    be a (real) power series. Then there exists a unique number \(R \in \ccintv{0}{\plusinfty}\), known as the \term{radius of convergence} of the power series, such that
    \begin{enumerate}[label=(\alph*)]
        \item If \(\abs{x - a} < R\), then the series converges absolutely.
        \item If \(\abs{x - a} > R\), then the series diverges.
    \end{enumerate}

    In particular,
    \[
        R\inverse = \limsup_{n \to \infty} \abs{c_n}^{\flatfrac{1}{n}}
    \]

    Furthermore, if \(R > 0\), and \(0 < r < R\), then the series converges absolutely uniformly on \(\abs{x - a} \leq r\). In particular, the convergence of the series is uniform in \(\abs{x - a} \leq r\) and defines a continuous function defined on \(\abs{x - a} < R\).
\end{theorem}

\begin{proof}
    The first part of the theorem is in \textit{IA Analysis I}. For any \(r\) with \(0 < r < R\), by the first part, we know that the series converges absolutely when \(x = a + r\), i.e., the series \(\sum_{n = 0}^{\infty} \abs{c_n} r^n\) converges. It follows that for any \(\abs{x - a} \leq r\), we may bound the terms by
    \[
        \abs{c_n \pqty{x - a}^n} \leq \abs{c_n} \abs{x - a}^{n} \leq \abs{c_N} r^n
    \]
    so by the \autoref{thm:weierstrass-m-test} (Weierstrass M-test) with \(M_n = \abs{c_n} r^n\), the series converges absolutely uniformly on \(\abs{x - a} \leq r\). The rest of the theorem follows from \autoref{cor:uniform-convergence-preserves-continuity} (Uniform Convergence preserves Continuity).
\end{proof}

\begin{definition}[Locally Uniform Convergence]
    Let \(\functiondomain{g_n, g}{I}{\RR}\) where \(I = \oointv{a}{b}\). We say \(g_n \to g\) \term{locally uniform} on \(I\) if for all \(y \in I\), there is some \(r > 0\), such that \(g_n \to g\) uniformly on \(\abs{x - y} \leq r\).
\end{definition}

In other words, \autoref{thm:radius-of-convergence} (Radius of Convergence) shows that a power series converges \emph{locally} uniformly on its interval of convergence.

\begin{theorem}[Differentiation of Power Series]
    \label{thm:differentiation-of-power-series}%
    Let \(\sum_{n = 0}^{\infty} c_n \pqty{x - a}^n\) be a (real) power series with radius of convergence \(R > 0\). Set
    \[
        f\pqty{x} = \sum_{n = 0}^{\infty} c_n
    \]
    for \(\abs{x - a} < R\). Then we have the following.

    \begin{enumerate}[label=(\roman*)]
        \item The ``derived power series'' \(\sum_{n = 1}^{\infty} n c_n \pqty{x - a}^{n - 1}\) has the same radius of convergence \(R\).
        \item \(f\) is differentiable on \(\abs{x - a} < R\), with its derivative given by the term-by-term derived power series
        \[
            f'\pqty{x} = \sum_{n = 1}^{\infty} n c_n \pqty{x - a}^{n - 1}.
        \]
    \end{enumerate}
\end{theorem}

\begin{proof}
    We have
    \[
        \limsup_{n \to \infty} \abs{n c_n}^{\flatfrac{1}{n}} = \limsup_{n \to \infty} n^{\flatfrac{1}{n}} \abs{c_n}^{\flatfrac{1}{n}}.
    \]
    since recall \(\lim_{n \to \infty} n^{\flatfrac{1}{n}} = 1\). So the radius of convergence of the derived power series is identical to that of the original power series.

    Let the partial sums be
    \[
        f_n\pqty{x} = \sum_{j = 0}^{n} c_j \pqty{x - a}^j
    \]
    and we know this is differentiable, with
    \[
        f_n'\pqty{x} = \sum_{j = 1}^{n} j c_j \pqty{x - a}^{j - 1}.
    \]

    Then \(f_n'\) converges uniformly to the derived power series on any interval (neighbourhood) \(\abs{x - a} \leq r\) for any \(0 < r < R\), and \(f'_n\pqty{a} = c_1\) converges. Hence, by \autoref{thm:uniform-convergence-differentiability} (Uniform Convergence and Differentiability), we have \(f\) is differentiable with its derivative being the derived power series at any point \(x\):
    \[
        f'\pqty{x} = \sum_{n = 1}^{\infty} n c_n \pqty{x - a}^{n - 1}.
    \]
    by picking \(r > 0\) such that \(\abs{x - a} \leq r < R\).
\end{proof}

\begin{remark}
    By \autoref{thm:differentiation-of-power-series}, if a power series
    \[
        \sum_{n = 0}^{\infty} c_n \pqty{x - a}^n
    \]
    has radius of convergence \(R > 0\), then the function \(f\) defined by this power series on \(\abs{x - a} < R\) is infinitely differentiable. Moreover,
    \[
        f^{\pqty{k}}\pqty{x} = \sum_{n = k}^{\infty} n \pqty{n - 1} \cdots \pqty{n - k + 1} c_n \pqty{x - a}^{n - k}
    \]
    gives the \(k\)th derivative of \(f\) at \(x\).

    In other words, take \(x = a\), then the coefficients are given by
    \[
        c_k = \frac{f^{\pqty{k}}\pqty{a}}{k!}
    \]
    for \(k = 0, 1, \ldots\): if we know the function \(f\) defined by a power series, in an arbitrary small interval (neighbourhood) around \(a\), then it allows us to compute all the derivatives of \(f\) at \(a\), and hence the function in the whole radius of convergence.

    In other words, if we have another power series \(\sum_{k = 0}^{\infty} d_n \pqty{x - a}^n\) is another power series with radius of convergence \(R_1 > 0\), defining the function \(g\), and \(g = f\) in some interval (neighbourhood) around \(a\), then \(R_1 = R\) and \(g = f\) everywhere for \(\abs{x - a} < R\), i.e., \(f\) has the ``unique continuation'' property.
\end{remark}

\begin{definition}[Analytic Functions]
    The functions defined by convergent power series (with \(R > 0\)) every point in its domain are called \term{analytic functions}.
\end{definition}

\begin{remark}
    It is \emph{not true} that every infinitely differentiable function is analytic. Consider the function
    \[
        f\pqty{x} = \begin{cases}
            e^{-\flatfrac{1}{x^2}}, & x \neq 0, \\
            0, & x = 0.
        \end{cases}
    \]

    Then \(f\) is infinitely differentiable, but \(f^{\pqty{k}}\pqty{0} = 0\) for all \(k\), so \(f\) is not analytic.
\end{remark}
