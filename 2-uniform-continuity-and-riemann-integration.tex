\section{Uniform Continuity and Riemann Integration}

Given a function \(\functiondomain{f}{E}{\RR}\) and \(E \subseteq \RR\), and given \(x \in E\), we recall from \textit{IA Analysis I} and say \(f\) is \term{continuous} at \(x\), if for all \(\epsilon > 0\), there is some \(\delta = \delta\pqty{\epsilon, x} > 0\), such that
\[
    \abs{y - x} < \delta \implies \abs{f\pqty{y} - f\pqty{x}} < \epsilon
\]

\subsection{Uniform Continuity}

\begin{definition}[Uniform Continuity]
    A function \(\functiondomain{f}{E}{\RR}\) is said to be \term{uniformly continuous on \(E\)}, if for all \(\epsilon > 0\), there is some \(\delta = \delta \pqty{\epsilon} > 0\), such that for all \(x, y \in E\),
    \[  
        \abs{y - x} < \delta \implies \abs{f\pqty{y} - f\pqty{x}} < \epsilon.
    \]
\end{definition}

It follows directly from the definition that uniform continuity implies continuity. But continuity at each point does \emph{not} imply uniform continuity.

\begin{example}[Continuity does not imply Uniform Continuity]
    For the function
    \[
        \function{f}{\oointv{0}{1}}{\RR}{x}{\frac{1}{x}},
    \]
    clearly \(g\) is continuous at each \(x \in \oointv{0}{1}\). For any choice of \(\delta\), if \(x \in \oointv{0}{\delta}\) and \(y = \flatfrac{x}{2}\), then
    \[
        \abs{x - y} = \frac{x}{2} < \delta,
    \]
    while
    \[
        \abs{g\pqty{x} - g\pqty{y}} = \frac{1}{x} > 1.
    \]

    So \(f\) is not uniformly continuous.
\end{example}

\begin{example}[Continuity and Bounded does not imply Uniform Continuity]
    Consider the function
    \[
        \function{g}{\oointv{0}{1}}{\RR}{x}{\sin\pqty{\frac{1}{x}}}
    \]
    which is bounded and continuous. Then consider the sequence
    \[
        x_n = \frac{1}{2 n \pi}, \qand y_n = \frac{1}{\pqty{2n + \frac{1}{2}} \pi}
    \]
    then \(\abs{x_n - y_n} \to 0\), but
    \[
        \abs{g\pqty{x_n} - g\pqty{y_n}} = 1
    \]
    for all \(n\). So \(g\) is not uniformly continuous on \(\oointv{0}{1}\).
\end{example}

\begin{theorem}[Continuity on Closed, Bounded Interval implies Uniform Continuity]
    \label{thm:continuity-implies-uniform-continuity}%
    Let \(a, b \in \RR\) and \(a < b\). If \(\functiondomain{f}{\ccintv{a}{b}}{\RR}\) is continuous at every point in \(\ccintv{a}{b}\), then \(f\) is uniformly continuous on \(\ccintv{a}{b}\).
\end{theorem}

\begin{proof}
    We prove by contradiction. If the claim is false, then there is some \(\epsilon > 0\), such that no \(\delta > 0\) work. Taking \(\delta = \frac{1}{n}\), we can find a pair of points \(x_n, y_n \in \ccintv{a}{b}\) with \(\abs{x_n - y_n} < \flatfrac{1}{n}\), and yet \(\abs{f\pqty{x_n} - f\pqty{y_n}} \geq \epsilon\).

    By the Bolzano--Weierstrass theorem, there is some \(x \in \ccintv{a}{b}\) and a subsequence \(x_{n_k}\), such that \(x_{n_k} \to x\) as \(k \to \infty\). But then
    \[
        \abs{y_{n_k} - x} \leq \abs{y_{n_k} - x_{n_k}} + \abs{x_{n_k} - x} \leq \frac{1}{n_k} + \abs{x_{n_k} - x} \to 0
    \]
    so we also have \(y_{n_k} \to x\).

    So by sequential continuity, \(f\pqty{x_{n_k}}, f\pqty{y_{n_k}} \to f\pqty{x}\), but this contradicts with \(\abs{f\pqty{x_n}  f\pqty{y_n}} > \epsilon\).
\end{proof}

\subsection{Riemann Integration}

We review facts from \textit{IA Analysis I}. Let \(\functiondomain{f}{\ccintv{a}{b}}{\RR}\) be a bounded function. Say \(f\) is bounded by \(m \leq f\pqty{x} \leq M\). Let a \term{partition} be
\[
    \partition = \Bqty{a_0, a_1, a_2, \ldots, a_n}
\]
with \(a = a_0 < a_1 < a_2 < \cdots < a_n = b\).

We define the \term{upper and lower Darboux sums} as
\begin{align*}
    \upperSum\pqty{\partition, f} &= \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j}\sup_{\ccintv{a_j}{a_{j + 1}}} f, \\
    \lowerSum\pqty{\partition, f} &= \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j}\inf_{\ccintv{a_j}{a_{j + 1}}} f,
\end{align*}
so we then have
\[
    m \pqty{b - a} \leq \lowerSum\pqty{\partition, f} \leq \upperSum\pqty{\partition, f} \leq M \pqty{b - a}.
\]

Also recall that \term{refining} a partition lowers the upper sum and increases the lower sum. Motivated by this, we define the \term{upper and lower Riemann integrals} of \(f\) by
\begin{align*}
    \upperI \pqty{f} = \inf_{\partition} \upperSum\pqty{\partition, f}, \\
    \lowerI \pqty{f} = \sup_{\partition} \lowerSum\pqty{\partition, f}. \\
\end{align*}

One may check that \(\lowerI \pqty{f} \leq \upperI \pqty{f}\). We say \(f\) is \term{Riemann integrable} in \(\ccintv{a}{b}\) if \(\upperI\pqty{f} = \lowerI\pqty{f}\), and we denote this value, the \term{Riemann integral} of \(f\) over \(\ccintv{a}{b}\), as
\[
    I\pqty{f} = \int_{a}^{b} f\pqty{x} \dd{x} = \int_{a}^{b} f.
\]

Also recall the following theorem.

\begin{theorem}[Riemann Criterion]
    A bounded function \(\functiondomain{f}{\ccintv{a}{b}}{\RR}\) is Riemann-integrable if and only if for all \(\epsilon > 0\), there is a partition \(\partition = \Bqty{a_0, \ldots, a_n}\) such that \(\upperSum\pqty{\partition, f} - \lowerSum\pqty{\partition, f} < \epsilon\),
    i.e.,
    \[
        \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j} \pqty{\sup_{I_j} f - \inf_{I_j} f} < \epsilon  
    \]
    where \(I_j = \ccintv{a_j}{a_{j + 1}}\).
\end{theorem}

\begin{theorem}[Composition of Integrable and Continuous Function is Integrable]
    Suppose \(\functiondomain{f}{\ccintv{a}{b}}{\ccintv{A}{B}}\) be bounded and integrable, and let \(\functiondomain{g}{\ccintv{A}{B}}{\RR}\) be continuous. Then \(\functiondomain{g \circ f}{\ccintv{a}{b}}{\RR}\) is integrable.
\end{theorem}

\begin{proof}
    Let \(\epsilon > 0\) be arbitrary. Since \(g\) is continuous on the closed interval \(\ccintv{A}{B}\), it is uniformly continuous by \autoref{thm:continuity-implies-uniform-continuity}. So there is some \(\delta = \delta\pqty{\epsilon} > 0\) such that for all \(x, y \in \ccintv{A}{B}\),
    \[
        \abs{x - y} < \delta \implies \abs{g\pqty{x} - g\pqty{y}} < \epsilon.
    \]

    Let \(\epsilon' > 0\) be arbitrary. Since \(f\) is integrable, there is a partition \(\partition = \Bqty{a_0, \ldots, a_n}\) of \(\ccintv{a}{b}\) such that
    \[
        \upperSum \pqty{\partition, f} - \lowerSum \pqty{\partition, f} = \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j} \pqty{\sup_{I_j} f - \inf_{I_j} f} < \epsilon'.
    \]
    
    Now, we investigate
    \[
        \upperSum\pqty{\partition, g \circ f} - \lowerSum\pqty{\partition, g \circ f} = \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j} \pqty{\sup_{I_j} \pqty{g \circ f} - \inf_{I_j} \pqty{g \circ f}}.
    \]

    We consider splitting this sum up. Let \(J\) be the set of indices
    \[
        J = \set{j \in \Bqty{0, \ldots, n - 1}}{\sup_{I_j} f - \inf_{I_j} f < \delta\pqty{\epsilon}}.
    \]

    For any \(j \in J\) and \(x, y \in I_j\), we must have
    \[
        \abs{f\pqty{x} - f\pqty{y}} \leq \sup_{I_j} f - \inf_{I_j} f < \delta\pqty{\epsilon},
    \]
    and thus
    \[
        \abs{g\pqty{f\pqty{x}} - g\pqty{f\pqty{y}}} < \epsilon.
    \]

    Taking the supremum over \(x \in I_j\) and the infimum over \(y \in I_j\), this implies that
    \[
        \sup_{I_j} \pqty{g \circ f} - \inf_{I_j} \pqty{g \circ f} < \epsilon.
    \]

    This implies that
    \[
        \sum_{j \in J} \pqty{a_{j + 1} - a_j} \pqty{\sup_{I_j} \pqty{g \circ f} - \inf_{I_j} \pqty{g \circ f}} < \epsilon \sum_{j \in J} \pqty{a_{j + 1} - a_j} \leq \epsilon\pqty{b - a}.
        \eqno{(1)}
    \]

    For \(j \notin J\), we have \(\sup_{I_j} f - \inf_{I_j} f \geq \delta\pqty{\epsilon}\), so we will need to bound the remaining part of the sum using some alternative method. We have
    \[
        \sum_{j \in J} \pqty{a_{j + 1} - a_j} \pqty{\sup_{I_j} f - \inf_{I_j} f} \leq 2 \sup_{\ccintv{a}{b}} \abs{g} \sum_{j \notin J} \pqty{a_{j + 1} - a_j}.
        \eqno{(2)}
    \]

    But on the other hand, we have
    \[
        \delta \sum_{j \notin J} \pqty{a_{j + 1} - a_j} \leq \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j} \pqty{\sup_{I_j} f - \inf_{I_j} f} < \epsilon'
        \eqno{(3)}
    \]
    by the integrability of \(f\).

    Given \(\epsilon\), choose \(\epsilon' = \delta\pqty{\epsilon} \epsilon\). Then we have by \((1), (2), (3)\), that
    \[
        \upperSum\pqty{\partition, g \circ f} - \lowerSum\pqty{\partition, g \circ f} < \pqty{\pqty{b - a} + 2 \sup_{\ccintv{a}{b}} \abs{g}} \epsilon.
    \]

    Then by the Riemann integrability criterion, we conclude that \(g \circ f\) is integrable.
\end{proof}

\begin{corollary}[Continuous Functions are Integrable]
    If \(\functiondomain{g}{\ccintv{a}{b}}{\RR}\) is continuous, then \(g\) is integrable.
\end{corollary}

\begin{proof}
    Take \(f = \identity\).
\end{proof}

We return to the discussion of uniform convergence and integrability.

\begin{theorem}[Uniform Limits of Integrable Functions are Integrable]
    Let \(\functiondomain{f_n}{\ccintv{a}{b}}{\RR}\) be a sequence of integrable functions, and suppose \(f_n \to f\) uniformly on \(\ccintv{a}{b}\). Then \(f\) is integrable, with \(\int_{a}^{b} f = \lim_{n \to \infty} \int_{a}^{b} f_n\).
\end{theorem}

\begin{proof}
    Since \(f_n \to f\) uniformly. Then for some \(N \in \NN\), we have a bound of
    \[
        \sup \abs{f_n - f} < 1
    \]
    for all \(n \geq N\). So by the triangular inequality, we have
    \[
        \sup \abs{f} \leq \sup \abs{f_N - f} + \sup \abs{f_N} \leq 1 + \sup \abs{f_N}
    \]
    so \(f\) is bounded.

    For any partition \(\partition = \Bqty{a_0, \ldots, a_n}\), we have
    \begin{align*}
        \upperSum\pqty{\partition, f} - \upperSum\pqty{\partition, f_n} &\leq \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j} \pqty{\sup_{I_j} f - \sup_{I_j} f_n} \\
        &= \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j} \bqty{\sup_{I_j} \pqty{f - f_n + f_n} - \inf_{I_j} \pqty{f - f_n + f_n}} \\
        &\leq \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j} \bqty{\sup_{I_j} \pqty{f - f_n} + \sup_{I_j} \pqty{f_n} - \inf_{I_j} \pqty{f - f_n} - \inf_{I_j} \pqty{f_n}} \\
        &\leq 2 \pqty{b - a} \sup_{\ccintv{a}{b}} \abs{f - f_n} + \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j} \bqty{\sup_{I_j} f_n - \inf_{I_j} f_n}.
    \end{align*}

    Given \(\epsilon > 0\), choose \(N\) such that
    \[
        \sup_{\ccintv{a}{b}} \abs{f - f_N} < \frac{\epsilon}{4 \pqty{b - a}}.
    \]

    Then, we choose a partition \(\partition\) such that
    \[
        \upperSum\pqty{\partition, f_N} - \lowerSum\pqty{\partition, f_N} = \sum_{j = 0}^{n - 1} \pqty{a_{j + 1} - a_j} \pqty{\sup_{I_j} f_N - \inf_{I_j} f_N} < \frac{\epsilon}{2}.
    \]  

    Then we have \(\upperSum\pqty{\partition, f} - \lowerSum\pqty{\partition, f} < \epsilon\), so \(f\) is integrable. Finally, we have from \autoref{thm:uniform-convergence-integrals} that the integrals converge.
\end{proof}

\subsection{Lebesgue's Criterion (Non-Examinable)}

\small

We further study conditions for Riemann integrability.

Clearly, a bounded, integrable function \(\functiondomain{f}{\ccintv{a}{b}}{\RR}\) need not be continuous everywhere. Let \(\mathcal{D}_f\) be the set of points of discontinuity of \(f\). How large does the set \(\mathcal{D}_f\) need to be for \(f\) to be non-integrable? We have the following result.

\begin{definition}[Null Set]
    A subset \(\mathcal{D} \subseteq \RR\) is said to bte a \term{null set} (or a set of \term{Lebesgue meausure zero}), if for all \(\epsilon > 0\), there is a countable collection of open intervals \(I_j = \oointv{a_j}{b_j}\) such that
    \[
        \mathcal{D} \subset \bigcup_{j = 1}^{\infty} I_j, \qand \sum_{j = 1}^{\infty} \pqty{b_j - a_j} < \epsilon.
    \]
\end{definition}

\begin{examples}
    \begin{enumerate}
        \item The empty set \(\emptyset\) and a singleton set \(\Bqty{a}\) are null sets.
        \item Any subset of a null set is a null set.
        \item A countable union of null sets is null.
        \item There are uncountable null sets, e.g., the Cantor set.
        \item Not every subset of \(\RR\) is null! Simply consider any interval \(I = \oointv{a}{b}\) with \(a < b\).
    \end{enumerate}
\end{examples}

\begin{lemma}
    If \(\oointv{a}{b} \subset \bigcup_{j = 1}^{\infty} I_j\), then \(\abs{b - a} \leq \sum_{j = 1}^{\infty} \pqty{b_j - a_j}\).
\end{lemma}

\begin{theorem}[Lebesgue's Criterion for Riemann Integrability]
    \label{thm:lebesgue-criterion-for-riemann-integrability}%
    A bounded function \(\functiondomain{f}{\ccintv{a}{b}}{\RR}\) is Riemann integrable if and only if the set of points of discontinuity \(\mathcal{D}_f\) is a null set.
\end{theorem}

\begin{corollary}
    \begin{enumerate}
        \item Monotone functions are integrable.
        \item If \(f\) is integrable and \(g\) is continuous, then \(g \circ f\) is integrable [since \(\mathcal{D}_{g \circ f} \subseteq \mathcal{D}_f\)].
        \item If \(f_n \to f\) uniformly and each \(f_n\) is integrable, then \(f\) is integrable [since \(\bigcup_{n = 1}^{\infty} \mathcal{D}_{f_n}\) is a null set].
   \end{enumerate}
\end{corollary}
