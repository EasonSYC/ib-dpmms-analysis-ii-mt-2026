\section{Metric Spaces and Normed Spaces}

So far, we have been doing analysis on the real line \(\RR\), and occasionally on the complex plane \(\CC\). \(\RR\) has a structure that allows us to define what is meant by
\begin{itemize}
    \item Convergence of sequences, with unique limits when they exist;
    \item Cauchy sequences, which always converge (the completeness property of \(\RR\));
    \item Closed bounded intervals \(\ccintv{a}{b}\) which are special due to the Bolzano-Weierstrass theorem.
\end{itemize}

With this structure, we studied functions \(\functiondomain{f}{E \subseteq \RR}{\RR}\), where we study
\begin{itemize}
    \item The most basic notion, continuity of a function;
    \item If \(E = \ccintv{a}{b}\) and \(f\) is continuous, then \(f\) is bounded and attains its bounds (the extreme value theorem, from \textit{IA Analysis I});
    \item Furthermore, in the same scenario, \(f\) is uniformly continuous.
\end{itemize}

We would like to extend these structures to a more general setting of functions \(\functiondomain{f}{X}{\RR}\), or even functions \(\functiondomain{f}{X}{Y}\). The easiest way to impose a structure on \(X\) is define a notion of distance between points in \(X\), which is known as \term{metric}. More specifically, when \(X\) is already a (real) vector space, through the notion of ``norm'', we can define a metric on \(X\) that is compatible with the vector space structure.

\subsection{Metric and Normed Spaces}

\begin{definition}[Metric Space]
    Let \(X\) be any set. A \term{metric} (or \term{distance function}) on \(X\) is a function \(\functiondomain{d}{X \times X}{\RR}\) satisfying, for any \(x, y, z \in X\),
    \begin{enumerate}[label=(\roman*)]
        \item \(d\) is \term{non-negative}: \(d\pqty{x, y} \geq 0\), and \(d\pqty{x, y} = 0\) if and only if \(x = y\);
        \item \(d\) is \term{symmetric}: \(d\pqty{x, y} = d\pqty{y, x}\);
        \item \(d\) satisfies the \term{triangle inequality}: \(d\pqty{x, z} \leq d\pqty{x, y} + d\pqty{y, z}\).
    \end{enumerate}

    The pair \(\pqty{X, d}\) is called a \term{metric space}.
\end{definition}

\begin{definition}[Normed Vector Space]
    Let \(V\) be a vector space over \(\RR\). A \term{norm} on \(V\) is a function \(\functiondomain{\norm{\cdot}}{V}{\RR}\) satisfying, for any \(x, y \in V\) and \(\lambda \in \RR\),
    \begin{enumerate}[label=(\roman*)]
        \item \(\norm{\cdot}\) is \term{non-negative}: \(\norm{x} \geq 0\), and \(\norm{x} = 0\) if and only if \(x = 0\);
        \item \(\norm{\cdot}\) is \term{absolutely homogeneous} (compatibility with scalar multiplication): \(\norm{\lambda x} = \abs{\lambda} \norm{x}\);
        \item \(\norm{\cdot}\) satisfies the \term{triangle inequality}:
        \[
        \norm{x + y} \leq \norm{x} + \norm{y}.
        \]
    \end{enumerate}

    The pair \(\pqty{V, \norm{\cdot}}\) is called a \term{normed vector space} (or simply a \term{normed space}).
\end{definition}

\begin{proposition}[Normed Space induces a Metric Space]
    \label{prop:normed-space-induces-metric-space}%
    If \(\pqty{V, \norm{\cdot}}\) is a normed space, and if \(\functiondomain{d}{V \times V}{\RR}\) is defined by \(d\pqty{x, y} = \norm{x - y}\), then \(\pqty{V, d}\) is a metric space.
\end{proposition}

\begin{proof}
    We check that this induced metric is indeed a metric.
    
    \begin{enumerate}[label=(\roman*)]
        \item This follows from the non-negativity of the norm, since \(d\pqty{x, y} = \norm{x - y} = 0\) if and only if \(x - y = 0\), i.e., \(x = y\).
        \item This follows from the absolute homogeneity of the norm, since
        \[
            d\pqty{x, y} = \norm{x - y} = \norm{-(y - x)} = \abs{-1} \norm{y - x} = \norm{y - x} = d\pqty{y, x}.
        \]
        \item This follows from the triangle inequality for norms. Note
        \begin{align*}
            d\pqty{x, z} &= \norm{x - z} \\
            &= \norm{\pqty{x - y} + \pqty{y - z}} \\
            &\leq \norm{x - y} + \norm{y - z} \\
            &= d\pqty{x, y} + d\pqty{y, z}.
        \end{align*}
    \end{enumerate}
\end{proof}

\begin{example}[Finite-Dimensional Normed Spaces]
    The prototypical finite-dimensional vector space is \(\RR^n\).
    \begin{enumerate}
        \item We equip it with the Euclidean (or \(\ell_2\)) norm. We define the norm to be
        \[
            \norm[\text{E}]{x} = \norm[\ell_2]{x} = \norm[2]{x} = \pqty{\sum_{i = 1}^{n} \abs{x_i}^2}^{\flatfrac{1}{2}}.
        \]

        The only non-trivial property to check is the triangular inequality. Let \(x = \pqty{x_1, \dots, x_n}\) and \(y = \pqty{y_1, \dots, y_n}\). Then
        \begin{align*}
            \norm[2]{x + y}^2 &= \sum_{i = 1}^{n} \pqty{x_i + y_i}^2 \\
            &= \norm[2]{x}^2 + \norm[2]{y}^2 + 2 \sum_{i = 1}^{n} x_i y_i \\
            &\leq \norm[2]{x}^2 + \norm[2]{y}^2 + 2 \norm[2]{x} \norm[2]{y} && \text{Cauchy--Schwarz Inequality}\\
            &= \pqty{\norm[2]{x} + \norm[2]{y}}^2.
        \end{align*}

        Taking the square root finishes the proof.

        \item Another norm is the \(\ell_1\) norm, defined by
        \[
            \norm[\ell_1]{x} = \norm[1]{x} = \sum_{i = 1}^{n} \abs{x_i}.
        \]

        \item Another norm is the \(\ell_\infty\) norm, defined by
        \[
            \norm[\ell_\infty]{x} = \norm[\infty]{x} = \sup_{1 \leq i \leq n}\abs{x_i}.
        \]
    \end{enumerate}

    The proofs to that the final two norms satisfying the triangle inequality are left as exercises to the reader. In general, \(\pqty{\RR^k, \norm[p]{\cdot}}\) is a normed space for any \(p \geq 1\).
\end{example}

\begin{example}[Space of Real-Number Sequences]
    Denote \(\RR^\NN\) be the space of real-number sequences, i.e.
    \[
        \RR^\NN = \Bqty{\text{sequences }\pqty{x_k}_{k \in \NN}\text{ with }x_k \in \RR}.
    \]

    Under term-wise addition and scalar multiplication, this is a vector space.

    \begin{enumerate}
        \item Define the subspace
        \[
            \ell_1 = \set{\pqty{x_k} \in \RR^\NN}{\sum_{k = 1}^{\infty} \abs{x_k} < \infty}.
        \]

        This is a vector subspace of \(\RR^\NN\), which is still infinite dimensional, and we define the \(\ell_1\) norm on the \(\ell_1\) space by
        \[
            \norm[1]{x} = \sum_{k = 1}^{\infty} \abs{x_k}.
        \]
        \item Similarly, we may define
        \[
            \ell_2 = \set{\pqty{x_k} \in \RR^\NN}{\sum_{k = 1}^{\infty} \abs{x_k}^2 < \infty}.
        \]

        We define the \(\ell_2\) norm (the Euclidean norm) on the \(\ell_2\) space by
        \[
            \norm[2]{x} = \pqty{\sum_{k = 1}^{\infty} \abs{x_k}^2}^{\flatfrac{1}{2}}.
        \]
        \item Finally, we may define
        \[
            \ell_\infty = \set{\pqty{x_k} \in \RR^\NN}{\sup_{k \in \NN} \abs{x_k} < \infty}
        \]
        to be the space of bounded sequences. We define the \(\ell_\infty\) norm on the \(\ell_\infty\) space by
        \[
            \norm[\infty]{x} = \sup_{k \in \NN} \abs{x_k}.
        \]
    \end{enumerate}

    \(\ell_1, \ell_2, \ell_\infty\) are all infinite-dimensional normed spaces. The proofs are left to the reader.

    There is nothing special about the \(1, 2, \infty\) norms above. For any \(p \geq 1\), we may define the \(\ell_p\) norm to be
    \[
        \ell_p = \set{\pqty{x_k} \in \RR^\NN}{\sum_{k = 1}^{\infty} \abs{x_k}^p < \infty},
    \]
    and equipped with the \(\ell_p\) norm,
    \[
        \norm[p]{x} = \pqty{\sum_{k = 1}^{\infty} \abs{x_k}^p}^{\flatfrac{1}{p}},
    \]
    the space \(\pqty{\ell_p, \norm[p]{\cdot}}\) is a normed space. Checking the triangular inequality is non-trivial and is left to the reader.
\end{example}

\begin{example}[Space of Continuous Functions]
    Consider the vector space \(V = \contClass{0} \pqty{\ccintv{a}{b}}\) to be the space of real-valued continuous function on \(\ccintv{a}{b}\).

    \begin{enumerate}
        \item The \(L^1\) norm on \(V\) is defined by
        \[
            \norm[L^1\pqty{\ccintv{a}{b}}]{f} = \norm[L^1]{f} = \norm[1]{f} = \int_{a}^{b} \abs{f\pqty{x}} \dd{x}.
        \]

        \item The \(L^2\) norm on \(V\) is defined by
        \[
            \norm[L^2\pqty{\ccintv{a}{b}}]{f} = \norm[L^2]{f} = \norm[2]{f} = \pqty{\int_{a}^{b} \abs{f\pqty{x}}^2 \dd{x}}^{\flatfrac{1}{2}}.
        \]

        \item The \(L^\infty\) norm, also known as the \term{uniform norm} on \(V\) is defined by
        \[
            \norm[L^\infty\pqty{\ccintv{a}{b}}]{f} = \norm[L^\infty]{f} = \norm[\infty]{f} = \sup_{x \in \ccintv{a}{b}} \abs{f\pqty{x}}.
        \]
    \end{enumerate}

    These all define norms on \(V\). The triangular inequality for \(L^2\) follows from the integral version of the Cauchy--Schwarz inequality
    \[
        \int_{a}^{b} \abs{fg} \leq \pqty{\int_{a}^{b} \abs{f}^2}^{\flatfrac{1}{2}} \pqty{\int_{a}^{b} \abs{g}^2}^{\flatfrac{1}{2}}.
    \]

    To prove this, we use
    \[
        \phi\pqty{t} = \int_{a}^{b} \pqty{\abs{g} - t \abs{f}}^2
    \]
    and note that this quadratic in \(t\) has to be non-negative. The degenerate case where the integral of \(f\) is zero is trivial.
\end{example}

By \autoref{prop:normed-space-induces-metric-space}, all of the above examples are metric spaces with respect to the metric defined by the norm. For example, the \term{Euclidean metric} in \(\RR^n\) is induced by the \(\ell_2\) norm, as
\[
    d_{\text{E}} \pqty{x, y} = \norm[2]{x - y} = \pqty{\sum_{i = 1}^{n} \abs{x_i - y_i}^2}^{\flatfrac{1}{2}}.
\]

The \term{uniform metric} on \(\contClass{0}\pqty{\ccintv{a}{b}}\) is induced by the uniform \(L^\infty\) norm, as
\[
    d\pqty{f, g} = \norm[\infty]{f - g} = \sup_{x \in \ccintv{a}{b}} \abs{f\pqty{x} - g\pqty{x}}.
\]

\small

The ``natural'' space in which the integral norms \(L^1, L^2, \ldots\) should be defined is
\[
    \mathcal{R} \pqty{\ccintv{a}{b}}
\]
the space of integrable functions over \(\ccintv{a}{b}\). This is a metric space with respect to pointwise addition and scalar multiplication of functions.

There is an immediate concern here: for \(f \in \mathcal{R} \pqty{\ccintv{a}{b}}\), \(\norm[1]{f} = 0\) does not necessarily imply \(f = 0\) on \(\ccintv{a}{b}\) -- it only implies that \(f = 0\) almost everywhere on \(\ccintv{a}{b}\) by \autoref{thm:lebesgue-criterion-for-riemann-integrability}, i.e., \(f = 0\) except on a null set.

If \(\int \abs{f} = 0\) then \(f = 0\) at the points of continuities, and hence \(f = 0\) except on a null set (which is at most the set of discontinuities).

On the other hand, if \(f = 0\) almost everywhere on \(\ccintv{A}{b}\), then \(\int \abs{f} = 0\) by covering the null set with intervals of arbitrarily small total length, which is allowed by definition.

This allows us to define an equivalence relation \(f \sim g\) on \(\mathcal{R} \pqty{\ccintv{a}{b}}\) by \(f \sim g\) if and only if \(f = g\) almost everywhere on \(\ccintv{a}{b}\), i.e., the set of points at which they differ is a null set. So we define the quotient space \(\tilde{\mathcal{R}}\pqty{\ccintv{a}{b}} = \flatfrac{\mathcal{R} \pqty{\ccintv{a}{b}}}{\sim}\), which would become a genuine normed space, and indeed we can show that the induced operations on the equivalence classes are well-defined, and that the norm is well-defined on the equivalence classes since the integrals are equal.

\normalsize

\begin{definition}[Discrete Metric]
    For any set \(X\), the \term{discrete metric} on \(X\) is given by
    \[
        \function{d}{X \times X}{\RR}{\pqty{x, y}}{
            \begin{cases}
                0, & \qfor x = y, \\
                1, & \qfor x \neq y.
            \end{cases}
        }
    \]
\end{definition}

\begin{example}[New Metric from Given Metrics]
    Let \(\pqty{X, d}\) be a metric space. Then a new induced metric can be defined as
    \[
        \function{g}{X \times X}{\RR}{\pqty{x, y}}{\min\Bqty{1, d\pqty{x, y}}.}
    \]

    Another metric could be defined by
    \[
        \function{h}{X \times X}{\RR}{\pqty{x, y}}{\frac{d\pqty{x, y}}{1 + d\pqty{x, y}}.}
    \]

    Note that \(g\pqty{x, y} \leq 1\) and \(h\pqty{x, y} < 1\) for any \(x, y \in X\).
\end{example}

\begin{example}[French Railway Metric]
    The SNCF metric (also known as the French railway metric) is defined on \(\RR^2\). The idea is ``all trains go through Paris'' (which will be the origin here). Define
    \[
        d\pqty{x, y} = \begin{cases}
            \norm[2]{x - y}, & \qif x = ty \text{ for some }t \in \RR,\\
            \norm[2]{x} + \norm[2]{y}, & \quad\text{otherwise}.
        \end{cases}
    \]
\end{example}

\begin{definition}[Metric Subspaces]
    Let \(\pqty{X, d}\) be a metric space, and let \(Y\) be a subset of \(X\). Then the restriction \(\rightbar{d}_{Y \times Y}\) is a metric on \(Y\). This is known as the \term{subspace metric} or the \term{induced metric} on \(Y\).
\end{definition}

\subsection{Open and Closed Subsets}

\begin{definition}[Open and Closed Subsets]
    \label{def:open-and-closed-subsets}%
    Let \(\pqty{X, d}\) be a metric space.
    \begin{enumerate}[label=(\roman*)]
        \item Let \(a \in X\) be any point, with \(r > 0\). Then the \term{open ball with radius \(r\) and centre \(a\)} is the set
        \[
            \oball{a}{r} = \set{x \in X}{d\pqty{x, a} < r}.
        \]

        \item A subset \(U \subseteq X\) is \term{open} if for every point \(a \in U\), there exists \(r > 0\) such that \(\oball{a}{r} \subseteq U\).
        \item A subset \(F \subseteq X\) is \term{closed} if its complement \(X \setminus F\) is open.
    \end{enumerate}
\end{definition}

\begin{remark}
    Note that sets can be both closed and open (\term{clopen}), for example \(\emptyset\) and \(X\), or neither closed nor open.
\end{remark}

\begin{examples}[Open and Closed Subsets in \(\RR^2\)]
    Consider \(X = \RR^2\).

    \begin{enumerate}[label=(\roman*)]
        \item Under the Euclidean metric, the open ball \(\oball[\text{E}]{a}{r}\) is the interior of a (Euclidean) circle with radius \(r\) and centre \(a\), as illustrated in \autoref{fig:open-ball-l2}.
        \item Under the \(\ell_{\infty}\) metric, the open ball \(\oball[\ell_{\infty}]{a}{r}\) is the interior of a square with side length \(2r\) and centre \(a\), as illustrated in \autoref{fig:open-ball-linfty}.
        \item Under the SNCF metric, the open ball \(\oball[\text{SNCF}]{0}{r}\) is simply the open ball \(\oball[\text{E}]{0}{r}\) under the Euclidean metric, as illustrated in \autoref{fig:open-ball-l2}. However, for \(a \neq 0\), the open ball \(\oball[\text{SNCF}]{a}{r}\) is the ray from the origin through \(a\) with radius \(r\) under the Euclidean metric, together with the open ball \(\oball[\text{E}]{0}{r - \norm[2]{a}}\) under the Euclidean metric (if \(r > \norm[2]{a}\)), as illustrated in \autoref{fig:open-ball-sncf-1} and \autoref{fig:open-ball-sncf-2}.
    \end{enumerate}

    The reason why the balls of the SNCF metric are so different from the Euclidean metric is that the SNCF metric is not translation invariant. In particular, all norm-induced metrics are translation invariant.
\end{examples}

Note that the property of being open or closed for a subset is relative to the containing metric space.

\begin{example}[Open and Closed Subsets in Subspaces]
    Let \(X = \RR\) with the Euclidean metric, and let \(Y = \cointv{0}{1} \cup \Bqty{2}\) with the induced metric from the Euclidean metric on \(\RR\).

    In \(Y\), we have the following.

    \begin{itemize}
        \item Note the set \(\cointv{0}{1}\) is both open and closed in \(Y\): in particular, \(\oball[Y]{0}{\flatfrac{1}{2}} = \cointv{0}{\flatfrac{1}{2}}\) is contained in \(\cointv{0}{1}\).
    
        \item Similarly, the singleton \(\Bqty{2}\) is both open and closed in \(Y\), by taking \(\oball[Y]{2}{\flatfrac{1}{2}} = \Bqty{2}\) to be contained in \(\Bqty{2}\).
    \end{itemize}
    
    However, in \(X = \RR\), things become a bit different.

    \begin{itemize}
        \item The set \(\cointv{0}{1}\) is neither closed nor open in \(X\).
        \item \(\Bqty{2}\) is closed in \(X\), but not open in \(X\).
    \end{itemize}
\end{example}

\begin{proposition}[Open Balls are Open and Singletons are Closed]
    \label{prop:open-balls-are-open-and-singletons-are-closed}%
    Let \(\pqty{X, d}\) be a metric space.

    \begin{enumerate}[label=(\roman*)]
        \item If \(a \in X\) and \(r > 0\), then the open ball \(\oball{a}{r}\) is an open subset of \(X\).
        \item For any \(a \in X\), the singleton \(\Bqty{a}\) is a closed subset of \(X\).
    \end{enumerate}
\end{proposition}

\begin{proof}
    \begin{enumerate}[label=(\roman*)]
        \item For any \(y \in \oball{a}{r}\), we have \(d\pqty{y, a} < r\). Let \(\epsilon = r - d\pqty{y, a} > 0\). Then for any \(z \in \oball{y}{\epsilon}\), we have
        \[
            d\pqty{z, a} \leq d\pqty{z, y} + d\pqty{y, a} < \epsilon + d\pqty{y, a} = r,
        \]
        so \(z \in \oball{a}{r}\). Hence, \(\oball{y}{\epsilon} \subseteq \oball{a}{r}\), so \(\oball{a}{r}\) is open. Visually, the relation between the open balls \(\oball{a}{r}\) and \(\oball{y}{\epsilon}\) is illustrated in \autoref{fig:open-ball-is-open}.

        \item For any \(x \in X \setminus \Bqty{a}\), we have \(d\pqty{x, a} > 0\). Let \(r = d\pqty{x, a}\), then \(\oball{x}{r} \subseteq X \setminus \Bqty{a}\), so \(X \setminus \Bqty{a}\) is open, and hence \(\Bqty{a}\) is closed, as illustrated in \autoref{fig:singleton-is-closed}.
    \end{enumerate}
\end{proof}

\begin{remark}
    An open ball can be closed, and a singleton can be open. For example, in the discrete metric space \(\pqty{X, d}\) where
    \[
        d\pqty{x, y} = \begin{cases}
            0, & \qif x = y,\\
            1, & \qif x \neq y,
        \end{cases}
    \]
    for any \(a \in X\), we have \(\oball{a}{r} = \Bqty{a}\) is both open and closed for all \(r \leq 1\).
\end{remark}

\begin{theorem}[Unions and Intersections of Open Sets]
    \label{thm:unions-and-intersections-of-open-sets}%
    Let \(\pqty{X, d}\) be a metric space.

    \begin{enumerate}[label=(\roman*)]
        \item The union of any collection of open sets in \(X\) is open.
        \item The intersection of any finite collection of open sets in \(X\) is open.
        \item \(\emptyset\) and \(X\) are both open and closed in \(X\).
    \end{enumerate}
\end{theorem}

\end{proof}

\begin{remark}
    The fact we restrict to finite intersections of open sets is important. For example, consider the open intervals \(\oball{0}{\flatfrac{1}{n}} = \oointv{-\flatfrac{1}{n}}{\flatfrac{1}{n}}\) for \(n \in \NN\). Then
    \[
        \bigcap_{n = 1}^{\infty} \oball{0}{\flatfrac{1}{n}} = \Bqty{0},
    \]
    which is not open in \(\RR\).
\end{remark}

By using de Morgan's laws, we can immediately deduce the following theorem.

\begin{theorem}[Unions and Intersections of Closed Sets]
    \label{thm:unions-and-intersections-of-closed-sets}%
    Let \(\pqty{X, d}\) be a metric space.

    \begin{enumerate}[label=(\roman*)]
        \item The intersection of any collection of closed sets in \(X\) is closed.
        \item The union of any finite collection of closed sets in \(X\) is closed.
        \item \(\emptyset\) and \(X\) are both open and closed in \(X\).
    \end{enumerate}
\end{theorem}

\begin{corollary}[Finite Sets are Closed]
    Let \(\pqty{X, d}\) be a metric space. Then any finite set \(F \subseteq X\) is closed.
\end{corollary}

\begin{proof}
    Any finite set is a finite union of singletons, and each singleton is closed by \autoref{prop:open-balls-are-open-and-singletons-are-closed} (ii), so the finite union is closed by \autoref{thm:unions-and-intersections-of-closed-sets} (ii).
\end{proof}

\subsection{Convergence in Metric Spaces}

\begin{definition}[Convergence in Metric Spaces]
    Let \(\pqty{X, d}\) be a metric space. A sequence \(\pqty{x_k}\) in \(X\) \term{converges to \(x \in X\)} if \(d\pqty{x_k, x} \to 0\) as \(k \to \infty\).

    Equivalently, \(\pqty{x_k}\) converges to \(x\) if
    \[
        \forall \epsilon > 0, \exists N \in \NN, \forall k \geq N: d\pqty{x_k, x} < \epsilon,
    \]
    or equivalently,
    \[
        \forall \epsilon > 0, \exists N \in \NN, \forall k \geq N: x_k \in \oball{x}{\epsilon}.
    \]
\end{definition}

\begin{proposition}[Limits are Unique in Metric Spaces]
    \label{prop:limits-are-unique-in-metric-spaces}%
    Let \(\pqty{X, d}\) be a metric space. If a sequence \(\pqty{x_k}\) converges to both \(x\) and \(y\) in \(X\), then \(x = y\).
\end{proposition}

\begin{proof}
    For all \(\epsilon > 0\), there exists \(N_1, N_2 \in \NN\) such that for all \(k \geq N_1\), \(d\pqty{x_k, x} < \flatfrac{\epsilon}{2}\), and for all \(k \geq N_2\), \(d\pqty{x_k, y} < \flatfrac{\epsilon}{2}\). Let \(N = \max\Bqty{N_1, N_2}\). Then for all \(k \geq N\),
    \[
        d\pqty{x, y} \leq d\pqty{x, x_k} + d\pqty{x_k, y} < \frac{\epsilon}{2} + \frac{\epsilon}{2} = \epsilon.
    \]

    This holds for arbitrary \(\epsilon > 0\), so \(d\pqty{x, y} = 0\), and hence \(x = y\).
\end{proof}

\begin{proposition}[Convergence in \(\RR^n\) with the Euclidean Metric is Componentwise]
    \label{prop:convergence-in-rn-is-componentwise}%
    Let \(\RR^n\) be equipped with the Euclidean metric, and
    let \(\pqty{\vb{x}^{\pqty{k}}}\) be a sequence in \(\RR^n\), where \(\vb{x}^{\pqty{k}} = \pqty{x_1^{\pqty{k}}, x_2^{\pqty{k}}, \ldots, x_n^{\pqty{k}}}\).
    
    Then \(\vb{x}^{\pqty{k}} \to \vb{x} = \pqty{x_1, \dots, x_n}\) in \(\RR^n\), if and only if \(x_{i}^{\pqty{k}} \to x_i\) in \(\RR\) for all \(i = 1, \dots, n\).
\end{proposition}

\begin{proof}
    \(\implies\). Fix \(\epsilon > 0\). Then there is some \(K \in \NN\) such that for all \(k \geq K\), we have \(\norm[2]{\vb{x}^{\pqty{k}} - \vb{x}} < \epsilon\).

    So we have
    \[
        \abs{x_j^{\pqty{k}} - x_j} \leq \norm[2]{\vb{x}^{\pqty{k}} - \vb{x}} < \epsilon,
    \]
    for all \(j = 1, \dots, n\). Hence, \(x_j^{\pqty{k}} \to x_j\) in \(\RR\) for all \(j = 1, \dots, n\).

    \(\impliedby\). For each \(j \in \Bqty{1, 2, \ldots, n}\), there is some \(K_j \in \NN\), such that for all \(k \geq K_j\), we habe \(\abs{x_j^{\pqty{k}} - x_j} < \flatfrac{\epsilon}{\sqrt{n}}\).

    Set \(K = \max{K_1, K_2, \ldots, K_n}\). Then for all \(k \geq K\), we have
    \[
        \norm[2]{\vb{x}^{\pqty{k}} - \vb{x}}^2 = \sum_{j = 1}^{n} \abs{x_j^{\pqty{k}} - x_j}^2 < n \cdot \frac{\epsilon^2}{n} = \epsilon^2,
    \]
    so \(\norm[2]{\vb{x}^{\pqty{k}} - \vb{x}} < \epsilon\), as desired.
\end{proof}

\begin{remark}
    Later, we will see that any norm-induced metric on \(\RR^n\) (and indeed, on any finite-dimensional vector space), are equivalent to each other, in a certain very precise way. Once we have this, convergence being componentwise will extend to any finite-dimensional vector space.
\end{remark}

\begin{definition}[Bounded Subsets]
    Let \(\pqty{X, d}\) be a metric space. A subset \(E \subseteq X\) is said to be \term{bounded}, if there is some ball \(\oball{x}{R}\), such that \(E \subseteq \oball{x}{R}\).
\end{definition}

\begin{remark}
    Note that then for any \(b \in X\), we have \(E \subseteq \oball{b}{R + d\pqty{x, b}}\). Hence, the choice of centre of the ball is not important.
\end{remark}

\begin{theorem}[Bolzano--Weierstrass Theorem in \(\RR^n\)]
    \label{thm:bolzano-weierstrass}%
    Any bounded sequence in \(\RR^n\) has a convergent subsequence.
\end{theorem}

\begin{remark}
    We will show this for the Euclidean norm, but by the above remark, this will hold for any norm-induced metric on \(\RR^n\).
\end{remark}

\begin{proof}
    Suppose \(\pqty{\vb{x}^{\pqty{k}}}\) is bounded in \(\RR^n\). So there is some \(R > 0\), such that \(\norm[2]{\vb{x}^{\pqty{k}}} < R\) for all \(k \in \NN\). Then each component sequence \(\pqty{x_j^{\pqty{k}}}\) is bounded in \(\RR\) for \(j = 1, \dots, n\). Let \(\vb{x}^{\pqty{k}} = \pqty{x_1^{\pqty{k}}, x_2^{\pqty{k}}, \ldots, x_n^{\pqty{k}}}\).

    Applying the Bolzano--Weierstrass theorem in \(\RR\) to each component sequence iteratively gives a convergent subsequence of \(\pqty{\vb{x}^{\pqty{k}}}\) in \(\RR^n\).

    Alternatively, we can prove using induction on the dimension \(n\), by writing
    \[
        \vb{x}^{\pqty{k}} = \pqty{\vb{y}^{\pqty{k}}, x_n^{\pqty{k}}},
    \]
    where \(\pqty{\vb{y}^{\pqty{k}}}\) is a sequence in \(\RR^{n - 1}\) and \(\pqty{x_n^{\pqty{k}}}\) is a sequence in \(\RR\). Note we have
    \[
        \norm[2]{\vb{x}^{\pqty{k}}}^2 = \norm[2]{\vb{y}^{\pqty{k}}}^2 + \abs{x_n^{\pqty{k}}}^2.
    \]

    So both \(\pqty{\vb{y}^{\pqty{k}}}\) and \(\pqty{x_n^{\pqty{k}}}\) are bounded, in \(\RR^{n - 1}\) and in \(\RR\) respectively. So by induction hypothesis, there is a subsequence \(\pqty{\vb{y}^{\pqty{k_j}}}\) of \(\pqty{\vb{y}^{\pqty{k}}}\) which converges in \(\RR^{n - 1}\) in \(\ell_2\) norm.
    
    Applying the Bolzano--Weierstrass theorem in \(\RR\) to the subsequence \(\pqty{x_n^{\pqty{k_j}}}\) gives a further subsequence \(\pqty{x_n^{\pqty{k_{j_l}}}}\) which converges in \(\RR\). Then the corresponding subsequence \(\pqty{\vb{y}^{\pqty{k_{j_l}}}}\) converges in \(\RR^{n - 1}\), and hence the sequence \(\pqty{\vb{x}^{\pqty{k_{j_l}}}}\) converges in \(\RR^n\).
\end{proof}

\begin{remark}
    Finite dimensionality of \(\RR^n\) is necessary for \autoref{prop:convergence-in-rn-is-componentwise} and \autoref{thm:bolzano-weierstrass}.
\end{remark}

\begin{example}
    \label{ex:bounded-sequence-in-l-infty-with-no-convergent-subsequence}%
    Consider \(\pqty{\ell_{\infty}, \norm[\infty]{\cdot}}\). Then the sequence of sequences
    \[
        e^{\pqty{k}} = \pqty{0, 0, \ldots, 0, 1, 0, \ldots}
    \]
    where the \(1\) is in the \(k\)-th position, i.e., 
    \[
        e_j^{\pqty{k}} = \kronecker_{kj} = \begin{cases}
            1, & \qif j = k,\\
            0, & \qif j \neq k.
        \end{cases}
    \]

    Then note \(\norm[\infty]{e^{\pqty{k}}} = 1\), so the sequence is bounded. But there is no convergent subsequence: if there is a convergent subsequence \(e^{\pqty{k_j}} \to e = \pqty{c_1, e_2, \ldots} \in \ell_{\infty}\) such that \(\norm[\infty]{e^{\pqty{k_j}} - e} \to 0\), then for all \(l\), we have \(\abs{e_l^{\pqty{k_j}} - c_l} \to 0\) as \(j \to \infty\) by the definition of the \(\ell_{\infty}\) norm.

    But \(e_l^{\pqty{k_j}} = 0\) for all sufficiently large \(j\), so \(e_l = 0\) for all \(l\). But \(\norm[\infty]{e^{\pqty{k_j}} - e} = \norm[\infty]{e^{\pqty{k_j}}} = 1\) for all \(j\), so the sequence cannot converge to \(e\).
\end{example}

\begin{definition}[Limit Points, Isolated Points, Closure, Interior]
    Let \(\pqty{X, d}\) be a metric space. Let \(E \subseteq X\) be any subset.

    \begin{enumerate}[label=(\roman*)]
        \item A point \(x \in X\) is a \term{limit point} of \(E\) if there is a sequence \(\pqty{x_k}\) in \(E\) with \(x_k \neq x\) for all \(k\),  such that \(x_k \to x\) in \(X\).
        \item A point \(x \in X\) is an \term{isolated point} of \(E\) if \(x \in E\) and \(x\) is not a limit point of \(E\).
        \item The \term{closure} of \(E\) is defined by
        \begin{align*}
            \closure{E} &= \set{x \in X}{x = \lim_{k \to \infty} x_k \text{ for some sequence } \pqty{x_k} \subseteq E} \\
            &= E \cup \set{x \in X}{x \text{ is a limit point of } E}
        \end{align*}
        \item A point \(x \in E\) is an \term{interior point} of \(E\) if there exists \(r > 0\) such that \(\oball{x}{r} \subseteq E\).
        \item The \term{interior} of \(E\), denoted \(\interior{E}\), is the set of all interior points of \(E\).
    \end{enumerate}
\end{definition}

\begin{example}
    Let \(X = \RR\) with the Euclidean metric, and let \(E = \cointv{0}{1} \cup \Bqty{2}\). Then we have
    \begin{itemize}
        \item The limit points of \(E\) are \(\ccintv{0}{1}\), so \(\closure{E} = \ccintv{0}{1} \cup \Bqty{2}\).
        \item The only isolated point of \(E\) is \(2\).
        \item The interior points of \(E\) are \(\oointv{0}{1}\), so \(\interior{E} = \oointv{0}{1}\).
    \end{itemize}
\end{example}

\subsection{Cauchy Sequences and Complete Metric Spaces}

\begin{definition}[Cauchy Sequences in Metric Spaces]
    Let \(\pqty{X, d}\) be a metric space. A sequence \(\pqty{x_k}\) in \(X\) is a \term{Cauchy sequence} if for all \(\epsilon > 0\), there exists \(N \in \NN\) such that for all \(m, n \geq N\), we have
    \[
        d\pqty{x_m, x_n} < \epsilon.
    \]
\end{definition}

\begin{proposition}[Convergence Implies Cauchy and Cauchy Implies Bounded, Subsequential Convergence]
    \label{prop:convergence-implies-cauchy-and-cauchy-implies-bounded-subsequential-convergence}%
    We have the following.
    \begin{enumerate}[label=(\roman*)]
        \item If \(\pqty{x_n}\) is a convergent sequence in \(X\), then \(\pqty{x_n}\) is a Cauchy sequence.
        \item If \(\pqty{x_n}\) is a Cauchy sequence in \(X\), then \(\pqty{x_n}\) is bounded.
        \item If \(\pqty{x_n}\) is a Cauchy sequence in \(X\) and has a convergent subsequence \(x_{n_k} \to x\), then \(x_n \to x\).
    \end{enumerate}
\end{proposition}

\begin{proof}
    \begin{enumerate}[label=(\roman*)]
        \item If \(x_n \to x\), then \(d\pqty{x_n, x_m} \leq d\pqty{x_n, x} + d\pqty{x_m, x} \to 0\) as \(n, m \to \infty\), so \(x_n\) is Cauchy.
        \item If \(\pqty{x_n}\) is Cauchy, then there exists \(N \in \NN\) such that for all \(m, n \geq N\), we have \(d\pqty{x_m, x_n} < 1\). Then for all \(n \geq N\), we have \(d\pqty{x_n, x_N} < 1\). Consider
        \[
            R = \max\Bqty{d\pqty{x_1, x_N}, d\pqty{x_2, x_N}, \ldots, d\pqty{x_{N - 1}, x_N}}.
        \]

        Then \(x_{n + 1} \in \oball{x_N}{R + 1}\) for all \(n \in \NN\), so \(\pqty{x_n}\) is bounded.
        \item Given \(\epsilon > 0\), there exists \(N\) such that \(d\pqty{x_n, x_m} < \epsilon\) for all \(m, n \geq N\). Therefore, for \(K\) such that \(n_k \geq N\) for all \(k \geq K\), we have \(d\pqty{x_{n_k}, x} < \epsilon\) for all \(k \geq K\). In particular, \(d\pqty{x_{n_K}, x} < \epsilon\). Then for all \(n \geq N\), we have
        \[
            d\pqty{x_n, x} \leq d\pqty{x_n, x_{n_K}} + d\pqty{x_{n_K}, x} < \epsilon + \epsilon = 2\epsilon,
        \]
        and hence \(x_n \to x\).
    \end{enumerate}
\end{proof}

\begin{definition}[Complete Metric Spaces and Normed Spaces]
    A metric space \(\pqty{X, d}\) is \term{complete} if every Cauchy sequence in \(X\) converges to some element in \(X\).

    A normed space \(\pqty{V, \norm{\cdot}}\) is \term{complete} if the metric space induced by the norm is complete.
\end{definition}

\begin{theorem}[Completeness of \(\RR^n\)]
    \label{thm:completeness-of-rn}%
    \(\RR^n\) (equipped with the Euclidean metric) is complete.
\end{theorem}

\begin{remark}
    Once we have equivalence of norms later in the course, \(\RR^n\) or any finite dimensional space with any norm is complete.
\end{remark}

\begin{proof}
    Let \(\pqty{\vb{x}^{\pqty{k}}}\) be a Cauchy sequence in \(\RR^n\) with respect to the norm \(\norm[2]{\cdot}\). Then for all \(\epsilon > 0\), there is some \(N \in \NN\), such that for all \(n, m \geq N\), we have
    \[
        \norm[2]{\vb{x}^{\pqty{n}} - \vb{x}^{\pqty{m}}} < \epsilon.
    \]

    Write \(\vb{x}^{\pqty{k}} = \pqty{x_1^{\pqty{k}}, x_2^{\pqty{k}}, \ldots, x_n^{\pqty{k}}}\). Then for each \(j = 1, \ldots, n\), we have
    \[
        \abs{x_j^{\pqty{n}} - x_j^{\pqty{m}}} \leq \norm[2]{\vb{x}^{\pqty{n}} - \vb{x}^{\pqty{m}}} < \epsilon,
    \]
    so the component sequences are Cauchy in \(\RR\), and using completeness of \(\RR\) with respect to the Euclidean metric, we see that each component sequence satisfies \(x_j^{\pqty{k}} \to x_j \in \RR\). Then by \autoref{prop:convergence-in-rn-is-componentwise}, we have \(\vb{x}^{\pqty{k}} \to \vb{x}\) in \(\RR^n\), where \(\vb{x} = \pqty{x_1, x_2, \ldots, x_n}\).
\end{proof}

\begin{theorem}[Completeness of \(\contClass{0}\pqty{\ccintv{a}{b}}\)]
    \label{thm:completeness-of-contclass-0}%
    The space \(\pqty{\contClass{0}\pqty{\ccintv{a}{b}}, \norm[\infty]{\cdot}}\) is complete.
\end{theorem}

\begin{proof}
    Let \(\pqty{f_k}\) be Cauchy with respect to the \(\ell_{\infty}\) norm. Then by \autoref{thm:cauchy-criterion-uniform-convergence} (Cauchy criterion), \(\pqty{f_k}\) converges in the \(\ell_{\infty}\) norm to some \(\functiondomain{f}{\ccintv{a}{b}}{\RR}\). By \autoref{cor:uniform-convergence-preserves-continuity}, \(f\) is continuous. 
\end{proof}

\begin{theorem}[Completeness of \(\pqty{\ell_p, \norm[p]{\cdot}}\)]
    \label{thm:completeness-of-ellp}%
    The spaces \(\pqty{\ell_p, \norm[p]{\cdot}}\) for \(p = 1, 2, \infty\).
\end{theorem}

\begin{proof}
    \(\ell_1\) and \(\ell_2\): \textbf{\textcolor{red}{See Example Sheet.}}

    To see that \(\pqty{\ell_{\infty}, \norm[\infty]{\cdot}}\) is complete, recall that \(\ell_{\infty}\) is the space of bounded functions \(\NN \to \RR\). So a Cauchy sequence in \(\ell_{\infty}\) is a Cauchy sequence of bounded functions with respect to the uniform norm of the function, which converges to a function by \autoref{thm:cauchy-criterion-uniform-convergence} (Cauchy criterion). \(f\) will be bounded (hence belongs to \(\ell_{\infty}\)) by the triangular inequality: since \(f_k \to f\), take \(\epsilon = 1\), then
    \[
        \norm[\infty]{f} \leq \norm[\infty]{f - f_k} + \norm[\infty]{f_k} < 1 + \norm[\infty]{f_k},
    \]
    for sufficiently large \(k\).
\end{proof}

\begin{theorem}[Subspace is Complete if and only if Closed]
    \label{thm:subspace-is-complete-if-and-only-if-closed}%
    Let \(\pqty{X, d}\) be a complete metric space. Let \(Y \subseteq X\) be equipped with the induced metric \(\rightbar{d}_{Y \times Y}\). Then \(Y\) is complete if and only if \(Y\) is closed in \(X\).
\end{theorem}

\begin{proof}
    \(\implies\). Suppose \(Y\) is complete. Let \(\pqty{y_k}\) be a sequence in \(Y\) with \(y_k \to x \in X\). Then \(\pqty{y_k}\) is Cauchy in \(X\), and hence in \(Y\). Since \(Y\) is complete, \(\pqty{y_k}\) converges to some \(y \in Y\). Limits are unique, so \(x = y \in Y\), and hence \(Y\) is closed in \(X\) by \autoref{prop:characterisation-of-closed-sets}.

    \(\impliedby\). Suppose \(Y\) is closed. Let \(\pqty{y_k}\) be a Cauchy sequence in \(Y\). Then \(\pqty{y_k}\) is Cauchy in \(X\), and since \(X\) is complete, there exists \(x \in X\) such that \(y_k \to x\). Since \(Y\) is closed, we have \(x \in Y\), and hence \(\pqty{y_k}\) converges in \(Y\).
\end{proof}

\begin{example}
    \(X = \oointv{0}{1}\) with the Euclidean metric is not complete. For example, the sequence \(x_k = \flatfrac{1}{k}\) is Cauchy in \(X\), but converges to \(0 \notin X\).

    Alternatively, \(X\) is not closed in \(\RR\), which is complete under the Euclidean metric, so \(X\) is not complete by \autoref{thm:subspace-is-complete-if-and-only-if-closed}.
\end{example}

\begin{example}
    Consider \(X = \RR \setminus \Bqty{0}\) with the Euclidean metric, and \(Y = \oointv{0}{\plusinfty}\). \(Y\) is closed in \(X\), since \(X \setminus Y = \oointv{\minusinfty}{0}\) is open in \(X\). But neither \(X\) nor \(Y\) is complete, since they are not closed in \(\RR\).

    Let \(Y_1 = \cointv{1}{\plusinfty}\). Then \(Y_1\) is complete, since \(Y_1\) is closed in \(\RR\), and it is closed in \(X\) as well. Therefore, a subspace of a non-complete metric space can be complete.
\end{example}

\subsection{Sequential Compactness and the Heine--Borel Theorem}

\begin{definition}[Sequential Compactness]
    \begin{enumerate}[label=(\roman*)]
        \item A metric space \(\pqty{X, d}\) is \term{sequentially compact} if every sequence in \(X\) has a convergent subsequence.
        \item A subset \(K \subseteq X\) is \term{sequentially compact} if every sequence in \(K\) has a convergent subsequence which converges to a point in \(K\).
    \end{enumerate}
\end{definition}

\begin{remark}
    Sequential compactness arises from the definition of compactness in general topological spaces, which would be covered in \textit{IB Topological Spaces}. Metric spaces are topological spaces, and for metric spaces, sequential compactness is equivalent to compactness. For this reason, we will use the term compactness in this course from now on.
\end{remark}

\begin{example}
    \begin{enumerate}[label=(\roman*)]
        \item \(I = \ccintv{0}{1}\) is a compact metric space with respect to the Euclidean metric. This is a consequence of the one-dimensional Bolzano--Weierstrass theorem.
        \item \(J = \oointv{0}{1}\) is not compact. The bounded sequence \(x_k = \flatfrac{1}{k}\) has no convergent subsequence in \(J\), since this sequence converges to \(0 \notin J\).
        \item The real line \(\RR\) is not compact. The sequence \(x_k = k\) has no convergent subsequence in \(\RR\).
    \end{enumerate}
\end{example}

\begin{theorem}[Compact Metric Spaces are Complete and Bounded]
    \label{thm:compact-metric-spaces-are-complete-and-bounded}%
    If \(\pqty{X, d}\) is a compact metric space, then \(X\) is complete and bounded.
\end{theorem}

\begin{proof}
    To prove completeness, let \(\pqty{x_k}\) be a Cauchy sequence in \(X\). By compactness, there is a subsequence \(\pqty{x_{k_j}}\) and some \(x \in X\), such that \(x_{k_j} \to x\). Then by \autoref{prop:convergence-implies-cauchy-and-cauchy-implies-bounded-subsequential-convergence} (iii), we have \(x_k \to x\) in \(X\), and hence \(X\) is complete.

    To prove bounded, fix \(x_0 \in X\). We prove by contradiction. If \(X\) is not bounded, then we can find a sequence \(x_k \in X\) such that \(d\pqty{x_k, x_0} > k\). Then by compactness, there is subsequence \(x_{k_j}\) and some \(x \in X\), such that \(x_{k_j} \to x\). But then
    \begin{align}
        k_j & \leq d\pqty{x_{k_j}, x_0} \\
        & \leq d\pqty{x_{k_j}, x} + d\pqty{x, x_0} \\
        & \to d\pqty{x, x_0} \text{ as } j \to \infty,
    \end{align}
    which is a contradiction, since \(k_j \to \infty\) as \(j \to \infty\). Hence, \(X\) is bounded.
\end{proof}

\begin{theorem}[Compact Subsets of Metric Spaces are Closed and Bounded]
    \label{thm:compact-subsets-of-metric-spaces-are-closed-and-bounded}%
    If \(\pqty{X, d}\) is a metric space, and \(K \subseteq X\) is compact, then \(K\) is closed and bounded.
\end{theorem}

\begin{proof}
    The restriction of the metric \(d\) to \(K\) makes \(\pqty{K, \rightbar{d}_{K \times K}}\) a compact metric space. Then by \autoref{thm:compact-metric-spaces-are-complete-and-bounded} (Complete Metric Spaces are Complete and Bounded), \(K\) is complete and bounded.
    
    Since \(K\) is complete, it is closed in \(X\) by \autoref{thm:subspace-is-complete-if-and-only-if-closed} (Subspace is Complete if and only if it is Closed). Concretely, suppose \(\pqty{x_k} \subseteq K\) and \(x_k \to x \in X\). As \(K\) is compact, there is some subsequence \(x_{k_j} \to y \in K\). But limits are unique, so \(x = y \in K\), and hence \(K\) is closed in \(X\).
\end{proof}

The converse of each of above holds for \(\RR^n\) (with the standard Euclidean metric), but not in general metric spaces.

\begin{examples}[Closed and Bounded Subsets of Metric Spaces Need Not be Compact]
    \begin{enumerate}
        \item In \autoref{ex:bounded-sequence-in-l-infty-with-no-convergent-subsequence}, we saw that the sequence \(\pqty{e^{\pqty{k}}}\) is bounded in \(\ell_{\infty}\) has no convergent subsequence. Hence, the closed and bounded subset \(\Bqty{e^{\pqty{k}}}{k \in \NN} \subseteq \ell_{\infty}\) is not compact.
        \item Let \(X = \RR\) with the metric \(d\pqty{x, y} = \min\Bqty{1, \abs{x - y}}\). Then \(\pqty{X, d}\) is a complete and bounded metric space. But \(X\) is not compact, since the sequence \(x_k = k\) has no convergent subsequence in \(X\), since for any fixed \(x \in X\), we have \(d\pqty{x_k, x} = 1\) for all sufficiently large \(k\).
    \end{enumerate}
\end{examples}

In \(\RR^n\) with the standard Euclidean metric, we have the following result.

\begin{theorem}[Heine--Borel]
    \label{thm:heine-borel}%
    Let \(K \subseteq \RR^n\). Then \(K\) is compact if and only if \(K\) is closed and bounded.
\end{theorem}

\begin{proof}
    The only if direction is by \autoref{thm:compact-subsets-of-metric-spaces-are-closed-and-bounded} (Compact Subsets of Metric Spaces are Closed and Bounded).

    The if direction is simply \autoref{thm:bolzano-weierstrass} (Bolzano--Weierstrass theorem) in \(\RR^n\). If \(\pqty{\vb{x}_k} \subseteq L\), then as \(k\) is bounded, the sequence \(\pqty{\vb{x}_k}\) is bounded, and hence has a convergent subsequence by \autoref{thm:bolzano-weierstrass}. Since \(L\) is closed, the limit of the convergent subsequence is in \(L\), and hence \(L\) is compact.
\end{proof}

In general metric spaces, completeness with a stronger version of boundedness is equivalent to compactness.

\begin{definition}[Totally Bounded]
    \label{def:totally-bounded}%
    A metric space \(\pqty{X, d}\) is \term{totally bounded} if for all \(\epsilon > 0\), we have finitely many \(x_1, \ldots, x_n \in X\), such that
    \[
        X = \bigcup_{i = 1}^n \oball{x_i}{\epsilon}.
    \]
\end{definition}

\begin{theorem}[Compactness is Equivalent to Completeness and Total Boundedness]
    \label{thm:compactness-equivalent-to-completeness-and-total-boundedness}%
    A metric space \(\pqty{X, d}\) is compact if and only if it is complete and totally bounded.
\end{theorem}

\subsection{Continuous Mappings between Metric Spaces}

\begin{definition}[Continuous Map]
    \label{def:continuous-mappings-between-metric-spaces}%
    Let \(\pqty{X, d}\) and \(\pqty{X', d'}\) be metric spaces. A map \(\functiondomain{f}{X}{X'}\) is \term{continuous at a point \(x \in X\)} if
    \[
        \forall \epsilon > 0, \exists \delta > 0: d\pqty{x, y} < \delta \implies d'\pqty{f\pqty{x}, f\pqty{y}} < \epsilon.
    \]

    Equivalently, using the open ball notation, this definition is equivalent to
    \[
        \forall \epsilon > 0, \exists \delta > 0: \oball[d]{x}{\delta} \subseteq f\inverse\pqty{\oball[d']{f\pqty{x}}{\epsilon}}.
    \]

    We say \(f\) is \term{continuous on \(X\)} if it is continuous at every point \(x \in X\).
\end{definition}

We will not distinguish between \(d\) and \(d'\) in the notation for open balls, if (which is the case usually) the metric is clear from the context.

\begin{theorem}[Sequential Characterisation of Continuity]
    \label{thm:sequential-characterisation-of-continuity}%
    Let \(\pqty{X, d}\) and \(\pqty{X', d'}\) be metric spaces. A map \(\functiondomain{f}{X}{X'}\) is continuous at \(x \in X\), if and only if for all sequences \(\pqty{x_k} \subseteq X\) with \(x_k \to x\), we have \(f\pqty{x_k} \to f\pqty{x}\).
\end{theorem}

\begin{proof}
    Suppose \(f\) is continuous at \(x \in X\), take any sequence \(\pqty{x_k} \subseteq X\) with \(x_k \to x\). Let \(\epsilon > 0\) be arbitrary. By definition of continuity, there exists \(\delta > 0\) such that
    \[
        \oball{x}{\delta} \subseteq f\inverse\pqty{\oball{f\pqty{x}}{\epsilon}}.
    \]

    But then since \(x_k \to x\), there exists some \(K \in \NN\), such that
    \[
        x_k \in \oball{x}{\delta} \subseteq f\inverse\pqty{\oball{f\pqty{x}}{\epsilon}}.
    \]

    Hence, we have
    \[
        f\pqty{x_k} \in \oball{f\pqty{x}}{\epsilon},
    \]
    so
    \[
        d'\pqty{f\pqty{x_k}, f\pqty{x}} < \epsilon,
    \]
    for all \(k \geq K\), and hence \(f\pqty{x_k} \to f\pqty{x}\).

    If \(f\) is not continuous at \(x \in X\), then there exists \(\epsilon > 0\) such that for all \(\delta > 0\), we have
    \[
        \oball{x}{\delta} \nsubseteq f\inverse\pqty{\oball{f\pqty{x}}{\epsilon}}.
    \]

    For \(k = 1, 2, 3, \ldots\), take \(\delta = \flatfrac{1}{k}\). Then there exists \(x_k \in \oball{x}{\flatfrac{1}{k}}\) such that
    \[
        f\pqty{x_k} \notin \oball{f\pqty{x}}{\epsilon},
    \]
    so \(x_k \to x\) but \(f\pqty{x_k} \not\to f\pqty{x}\), as \(d'\pqty{f\pqty{x_k}, f\pqty{x}} \geq \epsilon\) for all \(k\).
\end{proof}

There are many equivalent characterisations of continuity on the whole space.

\begin{theorem}[Characterisation of Continuity]
    \label{thm:characterisation-of-continuity}%
    Suppose \(\pqty{X, d}\) and \(\pqty{X', d'}\) are metric spaces. Let \(\functiondomain{f}{X}{X'}\). Then the following are equivalent.
    \begin{enumerate}[label=(\roman*)]
        \item \(f\) is continuous on \(X\).
        \item \textbf{Sequential Characterisation of Continuity.} For all sequences \(\pqty{x_k} \subseteq X\) with \(x_k \to x\), we have \(f\pqty{x_k} \to f\pqty{x}\).
        \item \textbf{Open Set Characterisation of Continuity.} For all open sets \(V \subseteq X'\), the pre-image \(f\inverse\pqty{V}\) is open in \(X\).
        \item \textbf{Closed Set Characterisation of Continuity.} For all closed sets \(E \subseteq X'\), the pre-image \(f\inverse\pqty{E}\) is closed in \(X\).
    \end{enumerate}
\end{theorem}

\begin{proof}
    (i) \(\iff\) (ii) is by \autoref{thm:sequential-characterisation-of-continuity} (Sequential Characterisation of Continuity).

    (iii) \(\iff\) (iv) is by taking complements with the fact about pre-images of complements
    \[
        f\inverse\pqty{X' \setminus A} = X \setminus f\inverse\pqty{A}.
    \]

    (i) \(\implies\) (iii). Let \(V \subseteq X'\) be open. Let \(x \in f\inverse\pqty{V}\) be arbitrary, then \(f\pqty{x} \in V\). Since \(V\) is open, there is some \(\epsilon > 0\) such that \(\oball{f\pqty{x}}{\epsilon} \subseteq V\). Then, continuity of \(f\) at \(x\) means there is some \(\delta > 0\), such that
    \[
        \oball{x}{\delta} \subseteq f\inverse\pqty{\oball{f\pqty{x}}{\epsilon}} \subseteq f\inverse\pqty{V}.
    \]

    Hence, \(f\inverse\pqty{V}\) is open in \(X\).

    (iii) \(\implies\) (i). Let \(x \in X\) and fix \(\epsilon > 0\). Apply the assumption in (iii) with the open set \(V = \oball{f\pqty{x}}{\epsilon}\). So \(f\inverse\pqty{\oball{f\pqty{x}}{\epsilon}}\) is open in \(X\), with \(x \in f\inverse\pqty{\oball{f\pqty{x}}{\epsilon}}\). Hence, there is some \(\delta > 0\) such that
    \[
        \oball{x}{\delta} \subseteq f\inverse\pqty{\oball{f\pqty{x}}{\epsilon}},
    \]
    which is exactly the definition of continuity at \(x\).
\end{proof}

\begin{definition}[Uniform Continuity]
    \label{def:uniform-continuity}%
    Let \(\pqty{X, d}\) and \(\pqty{X', d'}\) be metric spaces. A map \(\functiondomain{f}{X}{X'}\) is \term{uniformly continuous} on \(X\), if
    \[
        \forall \epsilon > 0, \exists \delta > 0, \forall x, y \in X: d\pqty{x, y} < \delta \implies d'\pqty{f\pqty{x}, f\pqty{y}} < \epsilon.
    \]

    Equivalently, this means
    \[
        \forall \epsilon > 0, \exists \delta > 0: \forall x \in X, \oball[d]{x}{\delta} \subseteq f\inverse\pqty{\oball[d']{f\pqty{x}}{\epsilon}}.
    \]
\end{definition}

\begin{definition}[Lipschitz Continuity]
    \label{def:lipschitz-continuity}%
    Let \(\pqty{X, d}\) and \(\pqty{X', d'}\) be metric spaces. A map \(\functiondomain{f}{X}{X'}\) is \term{Lipschitz continuous} on \(X\), if there exists \(L \in \cointv{0}{\infty}\) such that for all \(x, y \in X\), we have
    \[
        d'\pqty{f\pqty{x}, f\pqty{y}} \leq L d\pqty{x, y}.
    \]
    
    The smallest such \(L\) is called the \term{Lipschitz constant} of \(f\), given by
    \[
        \Lip\pqty{f} = \sup_{\substack{x, y \in X \\ x \neq y}} \frac{d'\pqty{f\pqty{x}, f\pqty{y}}}{d\pqty{x, y}}.
    \]
\end{definition}

\begin{remark}
    Lipschitz continuity implies uniform continuity, which implies continuity. No reverse implication holds in general.
\end{remark}

\begin{examples}
    \begin{itemize}
        \item The function \(\functiondomain{f}{\oointv{0}{1}}{\RR}\) given by \(f\pqty{x} = \flatfrac{1}{x}\) is continuous but not uniformly continuous.
        \item The function \(\functiondomain{f}{\ccintv{0}{1}}{\RR}\) given by \(f\pqty{x} = \sqrt{x}\) is uniformly continuous but not Lipschitz continuous.
    \end{itemize}
\end{examples}

\begin{theorem}[Continuous Functions on Compact Metric Spaces]
    Let \(\pqty{X, d}\) be a compact metric space, and \(\pqty{X', d'}\) be any metric space. Suppose \(\functiondomain{f}{X}{X'}\) is continuous. Then we have the following.

    \begin{enumerate}[label=(\roman*)]
        \item \(f\) is uniformly continuous.
        \item \textbf{Continuous image of Compact Set is Compact.} \(\img f = f\pqty{X} \subseteq X'\) is compact.
        \item \(f\pqty{X}\) is closed and bounded.
        \item \textbf{Extreme Value Theorem.} If \(X' = \RR\) with \(d'\) being the standard Euclidean metric, then \(f\) attains its supremum and infimum on \(X\).
    \end{enumerate}
\end{theorem}
